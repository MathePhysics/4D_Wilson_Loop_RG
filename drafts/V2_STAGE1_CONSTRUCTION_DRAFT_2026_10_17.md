\emph{Measurability and integrability.} The integrand of CRE (1.15) with the cutoff (1.25), in the variables $(V,V_0,A_{\mathrm e},a)$ of (1.22), is a finite composition of indicators of sets defined by inequalities between continuous functions (the tests), of the background $U_1(V,V_0)$, of the local minima $U_{1,\square'}(V)$ of (1.2) entering $\chi_0'$ and of the field $\mathbf H_{1,Ax}$ of (1.21) entering the bondwise maps (1.22), of CRE's analytic objects of (B4) and of the bondwise maps (1.22). The background is a Borel function of the admissible pairs, by the following argument, which uses (B1), (B2) and the smoothness of the constraint. Let $\mathcal O$ be the open set of fine configurations $U$ on which $\mathsf M(U)$ is defined on every row of $\Omega_1$, the local group means $\mathcal U_U(y,x)$, $x\in\mathcal B_y$, $y\in\Omega_1^{(1)}$, are defined (chart convention of the preamble; they are analytic there) and every $\alpha_c(U)$, $c\in\Omega_1^{[1]}$, is invertible (Section~\ref{sec:regular-hessian-stencils}), and let $\Phi(U)=(\bar U|_{\Omega_1^{[1]}},U|_{\Omega_1^c})$ on $\mathcal O$; $D\Phi(U)$ is onto, since $LQ_Uh_U=\id$ with $h_U$ supported on the designated bonds of $\Omega_1$, so each fibre $\Phi^{-1}(V,V_0)\cap\mathcal O$ is a $C^\infty$ submanifold and the tangent space of the fibre is the kernel of $Q_U$, the range of $\mathsf C_U$. Let $\mathrm{Crit}=\{U\in\mathcal O:D(S+\mathbf G(\Omega_1^{(1)},\cdot))(U)\circ\mathsf C_U=0\}$ be the set of critical points of CRE's constrained functional of (1.12) on the fibres; it is relatively closed in $\mathcal O$, being defined by the vanishing of a continuous function; the critical points of CRE's functional are read as the points of $\mathrm{Crit}$ (on $\mathcal O$ the fibres are manifolds, and $U_1\in\mathcal O$ by (B2) and its axial mean gauge). So the integrand is jointly Borel, and nonnegative. By Tonelli's theorem the section integral $\mathcal J^{\mathrm a}_\xi[1]=\int\mathcal K_\xi\,\dd a$ is a $[0,\infty]$-valued Borel function of the branch data, and $\int\mathcal O[\mathcal J^{\mathrm a}_\xi[1]](V)\,\dd V$ is the integral of the nonnegative integrand over all variables, equal to the fine integral $\mathcal Z_0$ of \eqref{eq:native-source-physical-ratio} below, which is finite (a bounded function on a compact group); hence the outer integrals converge, $\mathcal J^{\mathrm a}_\xi[1]<\infty$ for almost every value of the branch data, and on every branch $\mathcal J^{\mathrm a}_\xi[1]<\infty$, since the integrand \eqref{eq:native-source-actual-functional} is bounded on the bounded box $\{\Theta_\xi=1\}$ (its exponent is the sum of an analytic function bounded on the closed box, Lemma~\ref{lem:charts}, and of a constant of the branch, Lemma~\ref{lem:collar}). \subsection{The constrained Hessian at a regular background and its stencils}
\label{sec:regular-hessian-stencils}

Let $\mathsf M$ be Ba{\l}aban's averaging operation \eqref{eq:nonlinear-average} (written $M$ there; here $M$ is the cube side), and let $\Omega$ be a region, a union of blocks, and $U$ a configuration of the fine torus for which $\mathsf M(U)(c)$ is defined on every row $c\in\Omega^{[1]}$ of that region and the local group means $\mathcal U_U(y,x)$, $x\in\mathcal B_y$, are defined for every centre $y\in\Omega^{(1)}$ (chart convention of the preamble; for $U=U_1$ on $\Omega_1$ both hold, because $U_1\in D^{(11)}$ and $U_1$ is in its axial mean gauge, $\mathcal U_{U_1}(y,x)=\id$); the objects below are then defined for phases $q$ in a neighbourhood of zero depending on $U$, which Lemma~\ref{lem:two-block} and Appendix~\ref{app:background-estimates} quantify for the backgrounds that occur. The objects of this subsection are defined relative to the region $\Omega$: the rows are the coarse bonds $c\in\Omega^{[1]}$ (at least one block in $\Omega$), the block sums run over the centres $y\in\Omega^{(1)}$, the free carrier is $\Omega^*$, and the links of $U$ on bonds with both endpoints off $\Omega$ enter as fixed constants, with the phase variable equal to zero there. The torus itself ($\Omega=T$) is written without a superscript; the branch uses $\Omega=\Omega_1$ and $U=U_1$. Define, for $c\in\Omega^{[1]}$,
\begin{equation}\label{eq:native-source-constraint-normalisation}
    \mathcal Q_{U,c}(q)=\frac{1}{i}\log\left(\mathsf M(e^{iq}U)(c)\,\mathsf M(U)(c)^{-1}\right),\qquad
    Q_U=\frac{1}{L}D\mathcal Q_U(0),\qquad
    \mathcal N_U(q)=\mathcal Q_U(q)-LQ_Uq\;,
\end{equation}
so that $\mathcal N_U$ is the remainder denoted $\widetilde C$ in the linearisation following \cite[(2.10)]{BalabanRG1}; the logarithm is the group chart of the conventions above, defined when the relative average lies in the chart domain. For a coarse bond $c$ with designated fine bond $b_0(c)$, and a coarse centre $y$, put
\begin{equation}\label{eq:regular-stencil-bonds}
    \mathcal E_c=\{r\, | \,\{r_-,r_+\}\subset\mathcal B_{c_-}\cup\mathcal B_{c_+}\},
    \qquad
    \mathcal E_y=\{r\, | \,\{r_-,r_+\}\subset\mathcal B_y\}\;,
\end{equation}
$r$ ranging over the positive fine bonds.

\begin{lemma}[two-block structure]\label{lem:two-block}
Let $L\geq3$ be odd and let every coarse period exceed two. The row components $\mathcal Q_{U,c}$ and $\mathcal N_{U,c}$ depend only on the phases and background bonds in $\mathcal E_c$, and $b_0(c)$ is the only designated bond in $\mathcal E_c$. Hence
\begin{equation}\label{eq:regular-stencil-designated-diagonal}
    Q_{0,U}(c,c')=\delta_{cc'}\alpha_c(U),\qquad \alpha_c(U)=Q_U(c,b_0(c))\;,
\end{equation}
where $Q_{0,U}$ is the restriction of $Q_U$ to the designated bonds, with $\alpha_c(\id)=L^{-4}\id$. There is a phase radius $\tau_L>0$, depending on $L$ and the group only, such that for $U=e^{iA}$ with $\|A\|_\infty\leq\tau_L$ every $\alpha_c(U)$ is analytic in $A$ with smallest singular value at least $L^{-4}/2$. Under a site gauge transformation $u$, $\alpha_c(U^u)=\operatorname{Ad}_{u(c_-)}\alpha_c(U)\operatorname{Ad}_{u(b_0(c)_-)}^{-1}$.
\end{lemma}

\begin{proof}
Every path in the RG1 (0.4) row for $c$ lies in the two endpoint blocks, since the row averages parallel transports from the centre of $\mathcal B_{c_-}$ to the centre of $\mathcal B_{c_+}$ along paths inside the two blocks; a designated bond crosses its own coarse interface, and two adjacent blocks share exactly the interface labelled by $c$, so $b_0(c)$ is the only designated bond with both endpoints in $\mathcal B_{c_-}\cup\mathcal B_{c_+}$ (for $L\geq3$ odd the other designated bonds cross other interfaces). At the flat field, differentiating RG1 (0.4) gives \eqref{eq:covariance-two-averages} of Appendix~\ref{app:flat-covariance}: the straight-path term $(Q_1a)(c)=L^{-5}\sum_{x\in\mathcal B_{c_-}}a([x,x'])$ contains $b_0(c)$ in exactly $L$ of its $L^4$ translated straight paths of $L$ bonds each (the $L$ starting points on the line through $b_0(c)$), each with coefficient $L^{-5}$, and the mean-path term involves the paths $\Phi$ inside the blocks, which contain no designated bond; this gives $\alpha_c(\id)=L^{-4}\id$, the designated coefficient $Q_{0,\infty}=L^{-4}I$ computed in Appendix~\ref{app:flat-covariance} (paragraph ``Construction on the infinite lattice'', after \eqref{eq:bulk-operator-definition}; the computation is local and holds on every torus). For $U=e^{iA}$, $\alpha_c(U)$ is an analytic function of $A|_{\mathcal E_c}$ near zero (a composition of exponentials, the averaging and the chart logarithm) with $\alpha_c(\id)=L^{-4}\id$; the smallest singular value is continuous, so for each of the finitely many row types modulo $L\mathbb Z^4$ (translations commute with $\mathsf M$) there is a radius on which $\alpha_c$ is analytic with smallest singular value at least $L^{-4}/2$, and $\tau_L$ is the smallest of these radii; it depends on $L$ and the group only. The gauge covariance is $\mathsf M(U^u)(c)=u(c_-)\mathsf M(U)(c)u(c_+)^{-1}$ (RG1 (0.4) transforms at the block centres) together with $(R_uq)(b)=\operatorname{Ad}_{u(b_-)}q(b)$ for the phase (CRE (1.17)): $\mathcal Q_{U^u,c}(R_uq)=\operatorname{Ad}_{u(c_-)}\mathcal Q_{U,c}(q)$, hence $Q_{U^u}R_u=\operatorname{Ad}_{u(c_-)}Q_U$ row by row, and the $b_0(c)$ column gives the formula.
\end{proof}

By (B2) and Lemma~\ref{lem:two-block}, $\alpha_c(U)$ is invertible on every row of the regions used below. With $LQ_Uh_U=\id$ for the supported right inverse $h_U$, whose only nonzero outputs are at the designated bonds, the elimination maps are
\begin{equation}\label{eq:regular-stencil-elimination}
    (\mathsf C_UB)(b_0(c))
        =-\alpha_c(U)^{-1}\sum_{b\in\mathcal E_c\cap E_*}Q_U(c,b)B(b),
    \qquad
    (h_Uv)(b_0(c))=h_U(c)v(c),\qquad h_U(c)=\frac{1}{L}\alpha_c(U)^{-1}\;,
\end{equation}
and $(\mathsf C_UB)(r)=B(r)$ on free bonds $r\in E_*$; thus $Q_U\mathsf C_U=0$. The free-input and background stencils of a fine bond are
\begin{equation}\label{eq:regular-stencil-extension-support}
    (\mathcal F_r,\mathcal A_r)=
    \begin{cases}
        (\{r\},\varnothing),&r\in E_*\;,\\
        (\mathcal E_c\cap E_*,\mathcal E_c),&r=b_0(c)\;.
    \end{cases}
\end{equation}

The constrained quadratic form is CRE (1.14)--(1.16) on the region $\Omega$. With $J_U$ and $\Delta_U$ the first derivative and Hessian at $q=0$ of $q\mapsto S(e^{iq}U)$, $q\in\mathfrak g^{\Omega}$ (the plaquettes with a bond in $\Omega$ contribute; links off $\Omega$ are the fixed constants), and the relative gauge functional
\begin{equation}\label{eq:regular-covariance-gauge}
    \mathscr G^\Omega_U(q)=\sum_{y\in\Omega}\sum_{x\in\mathcal B_y\setminus\{y\}}
        \left(1-\operatorname{Re}\tr_N(\mathcal U_{e^{iq}U}(y,x)\mathcal U_U(y,x)^{-1})\right),
    \qquad G^{(2),\Omega}_U(a)=D^2\mathscr G^\Omega_U(0)[a,a]\;,
\end{equation}
where $\mathcal U_U(y,x)$ is the axial transport from the block centre $y$ to $x$ of CRE's gauge-fixing functional $\mathbf G$ in (1.11) and the sum is over the block centres of $\Omega$, the regional Hessian $\mathsf H^\Omega_U$ on $\mathfrak g^{\Omega^*}$ is the symmetric operator of the quadratic form ($B$ is extended by zero to the free bonds outside $\Omega^*$, so that $a=\mathsf C_UB$ is defined on every row of $\Omega^{[1]}$)
\begin{equation}\label{eq:regular-covariance-hessian}
    \langle B,\mathsf H^\Omega_UB\rangle
        =\langle a,\Delta_Ua\rangle+G^{(2),\Omega}_U(a)
            -\langle J_U,h_UD^2\mathcal N_U(0)[a,a]\rangle,
    \qquad a=\mathsf C_UB\;,
\end{equation}
with the rows of $\mathcal N_U$ and $h_U$ those of $\Omega^{[1]}$. The right side equals $D^2(S+\mathscr G^\Omega_U)(0)[a,a]-DS(0)[h_UD^2\mathcal N_U(0)[a,a]]$, which does not involve the inner product; $\mathsf H^\Omega_U$ is its operator for the fixed one, and its entries $\mathsf H^\Omega_U(b,b')\in\operatorname{End}(\mathfrak g)$ are those of the polarised bilinear form. This is CRE (1.16), $\langle A,\Delta_1A\rangle=\langle A,\Delta A\rangle-2\langle hC^{(2)}(A),J_1\rangle+\mathbf G^{(2)}(A)$, with CRE's $C^{(2)}(A)=\frac12D^2\mathcal N_U(0)[A,A]$ (the quadratic term of the linearising transformation, RG1 p.267, $\widetilde D^{(2)}=\widetilde C^{(2)}$) and $\mathbf G^{(2)}(A)=D^2\mathscr G(0)[A,A]$; at $\Omega=\Omega_1$, $U=U_1$ the relative gauge functional is CRE's absolute one, $\mathscr G^{\Omega_1}_{U_1}(q)=\mathbf G(\Omega_1^{(1)},e^{iq}U_1)$ with $\mathbf G$ of (1.7), (1.11), because $U_1$ is in its axial mean gauge, $\mathcal U_{U_1}(y,x)=\id$ for $x\in\mathcal B_y$, $y\in\Omega_1^{(1)}$ (Section~\ref{sec:branch-data}; CRE p.248 L36--38), so that $\mathbf G(\Omega_1^{(1)},U_1)=0$; and $\mathsf H^{\Omega_1}_{U_1}$ is CRE's $C^*\Delta_1C$ of (1.15), the same quadratic form written as an operator for the fixed inner product. The precision of the branch is
\begin{equation}\label{eq:c-regional-hessian}
    H_\xi=\mathsf H^{\Omega_1}_{U_1}\quad\text{on }\mathfrak g^{\Omega_1^*},\qquad H_{\xi,II}=\iota_I^*H_\xi\iota_I,\qquad H_{\xi,IJ}=\iota_I^*H_\xi\iota_J\;,
\end{equation}
with $\iota_I,\iota_J$ the extensions by zero. With $s_{p,U}(q)=1-\operatorname{Re}\tr_N((e^{iq}U)(\partial p))$, $\mathscr G_{y,U}$ the $y$-summand of \eqref{eq:regular-covariance-gauge}, and $a=\mathsf C_UB$, the form decomposes into local terms,
\begin{equation}\label{eq:regular-stencil-quadratic-decomposition}
    \langle B,\mathsf H^\Omega_UB\rangle=\sum_{p\cap\Omega\ne\varnothing}\mathfrak a_p(B;U)+\sum_{y\in\Omega}\mathfrak g_y(B;U)-\sum_{c\in\Omega^{[1]}}\mathfrak t_c(B;U),\qquad
    \begin{aligned}
        \mathfrak a_p&=D^2s_{p,U}(0)[a,a],\\
        \mathfrak g_y&=D^2\mathscr G_{y,U}(0)[a,a],\\
        \mathfrak t_c&=\tfrac1L\langle J_U(b_0(c)),\alpha_c(U)^{-1}D^2\mathcal N_{U,c}(0)[a,a]\rangle\;.
    \end{aligned}
\end{equation}
Their free-input and background sets are read off from Lemma~\ref{lem:two-block} and \eqref{eq:regular-stencil-elimination}: $a(r)$ depends on $B|_{\mathcal F_r}$ and on the background on $\mathcal A_r$, so $\mathfrak a_p$ has free-input set $\bigcup_{r\subset\partial p}\mathcal F_r$ and background set $\bigcup_{r\subset\partial p}(\{r\}\cup\mathcal A_r)$; $\mathfrak g_y$ involves the phases on the axial paths inside $\mathcal B_y$, which contain no designated bond (designated bonds cross interfaces), so its free-input and background sets are $\mathcal E_y\cap E_*$ and $\mathcal E_y$; $\mathfrak t_c$ involves $\mathcal N_{U,c}$, $\alpha_c$ and $J_U(b_0(c))$, the latter depending on the plaquettes through $b_0(c)$, which lie in $\mathcal B_{c_-}\cup\mathcal B_{c_+}$ for $L\geq3$, so its sets are $\mathcal E_c\cap E_*$ and $\mathcal E_c$. The entries of $\mathsf H^\Omega_U$ are the sums of the polarised entries of these terms. For a pair of positive bonds $(b,b')$ let $\mathcal T_{bb'}$ be the set of terms of \eqref{eq:regular-stencil-quadratic-decomposition} (plaquettes, centres, rows of the torus) whose free-input set contains both $b$ and $b'$, and let the \emph{entry stencil} $\mathcal S_{bb'}$ be the union of the background sets of these terms (one of finitely many types modulo $L\mathbb Z^4$; it is empty unless $d(b,b')\leq4L$, Lemma~\ref{lem:hessian-gauge-locality}). The entry function is
\begin{equation}\label{eq:c-entry-function}
    f_{bb'}(U)=\sum_{\mathfrak T\in\mathcal T_{bb'}}(\text{polarised entry of }\mathfrak T\text{ at }(b,b'))\in\operatorname{End}(\mathfrak g),\qquad f_{bb'}(A):=f_{bb'}(e^{iA})\;,
\end{equation}
defined for every configuration $U$ on $\mathcal S_{bb'}$ for which $\mathsf M(U)$ is defined and $\alpha_c(U)$ is invertible on the rows of the terms in $\mathcal T_{bb'}$ and the local group means $\mathcal U_U(y,x)$ are defined for the centres $y$ of the terms in $\mathcal T_{bb'}$ (the terms involve $\mathsf C_U$ on those rows and the means of those centres), whatever the configuration elsewhere; the torus form itself is not needed for its definition. The regional form is the compression of the torus form: for a background $U$ on which $\mathsf H^\Omega_U$ is defined (Section~\ref{sec:regular-hessian-stencils}: $\mathsf M(U)$ defined and $\alpha_c(U)$ invertible on the rows of $\Omega^{[1]}$, the means $\mathcal U_U(y,x)$ defined for $y\in\Omega^{(1)}$) and every pair $(b,b')\in\Omega^*\times\Omega^*$,
\begin{equation}\label{eq:c-regional-compression}
    \mathsf H^\Omega_U(b,b')=f_{bb'}(U|_{\mathcal S_{bb'}})\;,
\end{equation}
because a term whose free-input set contains the bond $b$ of $\Omega^*$ has its index in $\Omega$ (a plaquette containing $b$, or containing a bond $r$ with $b\in\mathcal F_r$, meets $\Omega$; a centre $y$ with $b\in\mathcal E_y$ lies in $\Omega$; a row $c$ with $b\in\mathcal E_c\cap E_*$ has an endpoint block containing an endpoint of $b$, so $c\in\Omega^{[1]}$), hence every term of $\mathcal T_{bb'}$ is a term of the regional sum, and the regional form is defined under the weaker requirement on $U$ stated above (Lemma~\ref{lem:c2-regional-entry} records this in the appendix).

\begin{lemma}[gauge law and locality of the Hessian]\label{lem:hessian-gauge-locality}
Let $u$ be a site gauge transformation of the torus, acting on the background by $U\mapsto U^u$ and on phases by $(R_uq)(b)=\operatorname{Ad}_{u(b_-)}q(b)$ (CRE (1.17)), and let $R_{u,*}$ be the restriction of $R_u$ to the free carrier $\Omega^*$; it is orthogonal for the invariant inner product. Then, for every region $\Omega$,
\begin{equation}\label{eq:regular-gauge-constrained-hessian}
    \mathsf H^\Omega_{U^u}=R_{u,*}\mathsf H^\Omega_UR_{u,*}^{-1}\;.
\end{equation}
Every entry $\mathsf H^\Omega_U(b,b')$ is a finite composition of local analytic operations in the background bonds of the entry stencil $\mathcal S_{bb'}$, which lies within the range $\rho_L=4L$ of $b$, including the fibre inverses $\alpha_c^{-1}$, and it vanishes for $d(b,b')>\rho_L$ (the stencils are listed in Appendix~\ref{app:background-estimates}, Lemma~\ref{lem:c2-entry-range}). The entry function $f_{bb'}$ of \eqref{eq:c-entry-function} is entrywise gauge covariant: for a site gauge transformation $u$,
\begin{equation}\label{eq:c-entry-gauge-covariance}
    f_{bb'}(U^u)=\operatorname{Ad}_{u(b_-)}f_{bb'}(U)\operatorname{Ad}_{u(b'_-)}^{-1}\;.
\end{equation}
The norms are the operator norm for the Hilbert norms induced by $h$ and, for $Df_{bb'}(A)$, the norm $\sup_{\|X\|_\infty\leq1}\|Df_{bb'}(A)[X]\|$. Let $\tau_L$ denote the radius of Lemma~\ref{lem:two-block}, reduced so that these operations are analytic on $\|A\|_\infty\leq\tau_L$ and reduced once more in \eqref{eq:c2-tau-final} of Appendix~\ref{app:background-estimates}; every statement is made for this final value. Define
\begin{equation}\label{eq:regular-covariance-entry-constants}
    m_L=\max_b\sum_{b'}\sup_{\|A\|_\infty\leq\tau_L}\|f_{bb'}(A)\|,
    \qquad
    c_L=\max_b\sum_{b'}\sup_{\|A\|_\infty\leq\tau_L}\|Df_{bb'}(A)\|\;.
\end{equation}
These depend on $L$, the group and the invariant inner product only.
\end{lemma}

\begin{proof}
The intertwinings are $\mathsf M(U^u)(c)=u(c_-)\mathsf M(U)(c)u(c_+)^{-1}$, hence $\mathcal Q_{U^u}\circ R_u=R^{(1)}_u\mathcal Q_U$ with $(R^{(1)}_uv)(c)=\operatorname{Ad}_{u(c_-)}v(c)$ (proof of Lemma~\ref{lem:two-block}), and therefore $Q_{U^u}R_u=R^{(1)}_uQ_U$, $\mathcal N_{U^u}\circ R_u=R^{(1)}_u\mathcal N_U$, $h_{U^u}=R_uh_U(R^{(1)}_u)^{-1}$ (from the covariance of $\alpha_c$ in Lemma~\ref{lem:two-block}) and $\mathsf C_{U^u}=R_u\mathsf C_UR_{u,*}^{-1}$. The three terms of \eqref{eq:regular-covariance-hessian} are then each preserved: $S((e^{iR_uq})U^u)=S((e^{iq}U)^u)=S(e^{iq}U)$ gives $\Delta_{U^u}=R_u\Delta_UR_u^{-1}$ and $J_{U^u}=R_uJ_U$; each relative mean $\mathcal U_{e^{iq}U}(y,x)\mathcal U_U(y,x)^{-1}$ is conjugated by $u(y)$, which preserves the trace; and $\langle J_{U^u},h_{U^u}D^2\mathcal N_{U^u}(0)[R_ua,R_ua]\rangle=\langle R_uJ_U,R_uh_UD^2\mathcal N_U(0)[a,a]\rangle$ equals the untransformed force term because $R_u$ is orthogonal for the invariant inner product. The same three identities hold term by term in \eqref{eq:regular-stencil-quadratic-decomposition} (each term is a function of $U$ and $a=\mathsf C_UB$ on its own stencil, and $\mathsf C_{U^u}R_{u,*}B=R_u\mathsf C_UB$), so the polarised entry of each term transforms as $\operatorname{Ad}_{u(b_-)}(\cdot)\operatorname{Ad}_{u(b'_-)}^{-1}$, which is \eqref{eq:c-entry-gauge-covariance}. Locality: the index of a term is the lower vertex $x(p)$ of its plaquette, the centre $y$ or the initial vertex of $b_0(c)$, and the distance of a bond to the index is that of its initial vertex. By the two-block structure of Lemma~\ref{lem:two-block} substituted into \eqref{eq:regular-stencil-quadratic-decomposition}, the free-input and background bonds of a block or row term have initial vertices in the one or two blocks of the index, within $\ell^\infty$ distance $2L-1$ of it (the sites of two adjacent blocks are within $2L-1$ of each other); a bond $r$ of $\partial p$ has initial vertex within $1$ of $x(p)$, and the bonds of $\mathcal F_r\cup\mathcal A_r$ are within $2L-1$ of $r$, so the bonds of a plaquette term are within $2L$ of its index. Hence a term whose free-input set contains $b$ and $b'$ has $d(b,b')\leq4L=\rho_L$, and its background bonds are within $4L$ of $b$. The stencils are listed in Lemma~\ref{lem:c2-entry-range}.
\end{proof}

\begin{lemma}[regular-background bounds]\label{lem:regular-bounds}
Let $\gamma_L=C_L^{-1}$ be the flat lower bound \eqref{eq:covariance-uniform-gap} of Appendix~\ref{app:flat-covariance}, $\tau_L$ the final radius of Lemma~\ref{lem:hessian-gauge-locality}, and let $R_{\mathrm c}$ (the collar range, Lemma~\ref{lem:collar-coverage}) and $C_{L,R_{\mathrm c}}$ be the constants fixed in Appendix~\ref{app:background-estimates}, Section~\ref{app:regular-covariance}; they depend on $L$ and the group only. Whenever \eqref{eq:c-standing-conditions} holds and $\widehat\mu=C_{L,R_{\mathrm c}}\varepsilon_1$, $\varepsilon_1=g_1p_0(g_1)$ (CRE p.246, with $g_1$ the first-step coupling), satisfies $\widehat\mu<\tau_L/2$ and $\max(1,c_L)\widehat\mu<\gamma_L/4$, on every branch (where $H_\xi$ is defined, by (B1) and (B2), as the operator of a quadratic form at fixed volume)
\begin{equation}\label{eq:c-regular-two-sided}
    \frac{\gamma_L}{2}\,\id\leq H_{\xi,II}\leq m_L\,\id\;,
\end{equation}
and $H_{\xi,II}(b,b')=0$ for $d(b,b')>\rho_L$ (the entry stencil $\mathcal S_{bb'}$ is empty beyond that distance, Lemma~\ref{lem:c2-entry-range}). The same bounds hold for every principal restriction of $H_{\xi,II}$.
\end{lemma}

The proof is Appendix~\ref{app:background-estimates}, Section~\ref{app:regular-covariance}: for $b,b'\in I$ the entries $(H_\xi)_{bb'}$ are $f_{bb'}(U_1|_{\mathcal S_{bb'}})$ by \eqref{eq:c-regional-compression} (Lemma~\ref{lem:c2-regional-entry}); a partition of unity on the coarse grid, a local gauge on each patch bringing $U_1$ to $e^{iA}$ with $\|A\|_\infty\leq\widehat\mu$ (Lemmas~\ref{lem:c2-plaquette-deviation} and~\ref{lem:c2-local-gauge}), and the entrywise fundamental theorem of calculus compare the form with the flat one.

\subsection{The source phase and the local primitives}
\label{sec:source-phase-primitives}

On the branch, put $U=U_1$ and $\Omega=\Omega_1$. The definitions and Lemma~\ref{lem:row-locality} of this subsection use only (B2), (B4), Lemma~\ref{lem:charts} (for the row solutions on the near rows), Section~\ref{sec:regular-hessian-stencils}, the map $\Psi_\xi$ of \eqref{eq:native-source-exact-boundary-map} with \eqref{eq:c-bondwise-shift} and \eqref{eq:c-collar-distance}, and Lemma~\ref{lem:collar-coverage}(i),(iii). They are stated for the background $U=U_1$ of the branch. For the reference background $U=\id$ of Section~\ref{sec:joint-extraction} only the near rows and the germs occur (the reference inputs vanish off $I$), the row solutions are those of Lemma~\ref{lem:charts} in its reference form (Lemma~\ref{lem:c2-reference}), (B4) is not invoked, and $v$, $V$ below are defined by the same formulas. For a free vector $B\in\mathfrak g^{\Omega_1^*}$ put $A=g_0\mathsf C_UB\in\mathfrak g^{\Omega_1}$. The first-step correction $\widetilde D_U$ of CRE (1.13)--(1.14) is the solution of the system
\begin{equation}\label{eq:native-source-local-implicit}
    \widetilde D_{U,c}(A)=\mathcal N_{U,c}\bigl(A-h_U\widetilde D_U(A)\bigr),\qquad c\in\Omega_1^{[1]}\;,
\end{equation}
the equation of RG1 p.267 ($L\widetilde QB'+\widetilde C(B')=L\widetilde QB$ with $B'=B-h\widetilde D(B)$, $\widetilde C=\mathcal N_U$, $\widetilde Q=Q_U$) written row by row. Let $\rho_{\mathrm d}>0$ be the radius $2K_*\vartheta_*^2$ of Appendix~\ref{app:background-estimates}, Lemma~\ref{lem:c2-uniform-chart}(ii); it depends on $L$ and the group only. A solution $d_c$ of the row equation of $c$ with $h(d_c)<\rho_{\mathrm d}$ is called \emph{small}. The row solution $\widetilde D_{U,c}(A)$ is defined as follows: for $U=U_1$ on the rows not within $2R_{\mathrm c}+2L$ of $I$ (the far rows; a row is within $\rho$ of a bond set when every bond of its two-block set is, see below) it is CRE's $\widetilde D_c(A)$ of (B4)(i), at the arguments $A=g_0\mathsf C_{U_1}B$, $B\in\mathcal A_\xi$, of the domain of integration \eqref{eq:c-free-variable-domain}; on the rows within $2R_{\mathrm c}+2L$ of $I$ (the near rows), for $U=U_1$ and for the reference background, it is the unique small solution of the row equation of $c$ at the argument $A|_{\mathcal E_c}$ given by Lemma~\ref{lem:charts} on the row polydisc $\mathcal U_c$; and on a fixed neighbourhood $\mathcal N_c$ of $A|_{\mathcal E_c}=0$, for every row, it is the germ of Lemma~\ref{lem:row-locality}, where $\mathcal N_c$ is the neighbourhood given by that lemma, reduced so that the germ takes values with $h<\rho_{\mathrm d}$ (it vanishes at zero). Where two of these definitions apply they agree, because each produces a small solution and the small solution is unique: on the near rows by Lemma~\ref{lem:charts}, and on every row by (B4)(i), RG1's ``exactly one solution''. The reconstructed phase is the function of the free vector
\begin{equation}\label{eq:native-source-local-phase}
    q_U(B)=A-h_U\widetilde D_U(A)\in\mathfrak g^{\Omega_1},\qquad A=g_0\mathsf C_UB\;,
\end{equation}
defined on free and designated bonds of $\Omega_1$ wherever every row solution is defined. The integration variable of the branch is $a+A_{\mathrm e}\in\mathfrak g^{\Omega_1^*}$ in the variables of CRE (1.22); with $\Psi_\xi$ the bondwise map \eqref{eq:native-source-exact-boundary-map} of Section~\ref{sec:boundary-law-collar} from these variables to those of (1.15), the reconstructed fine field is
\begin{equation}\label{eq:c-reconstructed-field}
    U_\xi(a)(b)=e^{iq_U(\Psi_\xi(a+A_{\mathrm e}))(b)}U_1(b)\quad(b\in\Omega_1),\qquad U_\xi(a)(b)=V_0(b)\quad(b\in\Omega_1^c)\;,
\end{equation}
the fine field $U=U'U_1$ of CRE (1.13) at the actual branch data, $U_\xi(a)=U(\Psi_\xi(a+A_{\mathrm e}))$ in the notation of Section~\ref{sec:branch-data}; it is defined for $(A_{\mathrm e},a)\in\mathcal D_\xi$ by the definition of $\mathcal D_\xi$ and $\mathcal A_\xi$, and by Lemma~\ref{lem:charts} and Lemma~\ref{lem:row-locality} its links on the rows within $2R_{\mathrm c}+2L$ of $I$ are defined for every $a$ in the enlarged box defined there, the other links not depending on $a$. \begin{equation}\label{eq:c-bondwise-shift}
    \Psi_\xi(a+A_{\mathrm e})=a+\Psi_\xi(A_{\mathrm e})\qquad(a\in\mathfrak g^I,\ A_{\mathrm e}\in\mathfrak g^J)\;.
\end{equation}
A row $c$ is said to lie within $\rho$ of a bond set $S$ when every bond of $\mathcal E_c$ is within distance $\rho$ of $S$; as the bonds of $\mathcal E_c$ have their initial vertices in two adjacent blocks, this holds as soon as one bond of $\mathcal E_c$ is within $\rho-(2L-1)$ of $S$.

\begin{lemma}[rowwise locality of the reconstruction]\label{lem:row-locality}
The equation of row $c$ in \eqref{eq:native-source-local-implicit} involves $\widetilde D_{U,c}$ alone and depends only on $A|_{\mathcal E_c}$ and $U|_{\mathcal E_c}$; so the system decouples row by row, and a solution $d_c$ of the row equation at the argument $A|_{\mathcal E_c}$ is a function of $A|_{\mathcal E_c}$ and $U|_{\mathcal E_c}$. The row equation has a unique analytic germ of solutions at $(A|_{\mathcal E_c},d_c)=(0,0)$, with $d_c(0)=0$ and $Dd_c(0)=0$; for $U=U_1$ it is the germ at zero of CRE's $\widetilde D_c$. Consequently, wherever the row solutions are defined (Section~\ref{sec:source-phase-primitives}): on a free bond $r$, $q_U(B)(r)=g_0B(r)$; on $b_0(c)$, $q_U(B)(b_0(c))$ depends only on $B|_{\mathcal E_c\cap E_*}$ and on $U|_{\mathcal E_c}$; and $q_U(B)$ vanishes on all of $\mathcal E_c$ whenever $B=0$ on $\mathcal E_c\cap E_*$ and the row solution is the germ value there. In particular, by \eqref{eq:c-bondwise-shift}, the link $U_\xi(a)(r)$ depends on $a$ only through $a|_{\mathcal F_r\cap I}$ for $r\in\Omega_1$, and not at all for $r\in\Omega_1^c$.
\end{lemma}

\begin{proof}
$\mathcal E_c$ contains no designated bond other than $b_0(c)$ (Lemma~\ref{lem:two-block}), so $h_U\widetilde D_U(A)$ restricted to $\mathcal E_c$ involves only $\widetilde D_{U,c}$, and $\mathcal N_{U,c}$ depends on its argument restricted to $\mathcal E_c$ (Lemma~\ref{lem:two-block}); by \eqref{eq:regular-stencil-elimination}, $A|_{\mathcal E_c}$ depends on $B|_{\mathcal E_c\cap E_*}$ only. At $(0,0)$ the derivative in $d_c$ of $d_c-\mathcal N_{U,c}(A-h_U(c)d_c)$ is the identity, since $D\mathcal N_U(0)=D\mathcal Q_U(0)-LQ_U=0$, and $\mathcal N_{U,c}(0)=0$ (the zero phase reproduces the averages of $U$). The implicit function theorem gives the germ, unique among continuous solutions near $(0,0)$, with $d_c(0)=0$ and $Dd_c(0)=D\mathcal N_{U,c}(0)(\ldots)=0$. CRE's $\widetilde D$ is analytic with $\widetilde D(0)=0$ and an expansion beginning with quadratic terms (RG1 p.267, (B4)), and its row $c$ solves the row equation (B4)(i), so its germ at zero is this one. If $B=0$ on $\mathcal E_c\cap E_*$ then $A=0$ on $\mathcal E_c$, so the germ value is $\widetilde D_{U,c}(0)=0$ and $q_U(B)=0$ on $\mathcal E_c$. The last claim follows from \eqref{eq:c-reconstructed-field}, \eqref{eq:c-bondwise-shift} and $\mathcal F_{b_0(c)}=\mathcal E_c\cap E_*$: the links of $U_\xi(a)$ on the bonds of $\mathcal E_c$ involve the row solution of $c$ at the argument $g_0\mathsf C_U\Psi_\xi(a+A_{\mathrm e})|_{\mathcal E_c}$, which involves $a$ only through $a|_{\mathcal E_c\cap E_*\cap I}$.
\end{proof}

The analyticity of the reconstruction on the cutoff box is the following lemma, proved in Appendix~\ref{app:background-estimates} (Lemmas~\ref{lem:c2-uniform-chart} and~\ref{lem:c2-primitive-bounds}); the enlarged box is $\{a\in\mathfrak g^I\, | \,|a(b)|\leq2r\ \forall b\in I\}$ with $r$ the cutoff radius of \eqref{eq:native-source-young-cutoff}, $R_{\mathrm c}$ is the collar range of Lemma~\ref{lem:collar-coverage}, the row polydisc of a row $c$ is $\mathcal U_c=\{B\in\mathfrak g^{\Omega_1^*}\, | \,|B(b)|<2+4r\ \forall b\in\mathcal E_c\cap E_*\}$, and the block polydisc of a centre $y$ is $\mathcal U_y=\{B\, | \,|B(b)|<2+4r\ \forall b\in\mathcal E_y\}$ (every bond of $\mathcal E_y$ is free). A centre $y$ lies within $\rho$ of a bond set $S$ when every bond of $\mathcal E_y$ does.

\begin{lemma}[charts]\label{lem:charts}
Let $\vartheta_*>0$ be the phase radius of Appendix~\ref{app:background-estimates}, Lemma~\ref{lem:c2-uniform-chart} ($\vartheta_*\leq\tau_L/2$; it depends on $L$ and the group only). On every branch, and for the reference background $U=\id$ of Section~\ref{sec:joint-extraction}, for every row $c\in\Omega_1^{[1]}$ within $2R_{\mathrm c}+2L$ of $I$ and every real phase $A$ with $\|A|_{\mathcal E_c}\|\leq\vartheta_*/2$ (Hilbert norm of $\mathfrak g^{\mathcal E_c}$), the row equation of $c$ at $A$ has exactly one small solution, $\widetilde D_{U,c}(A)=\varphi_c(A|_{\mathcal E_c})$, where $\varphi_c$ is analytic on the complex ball $\{\|A|_{\mathcal E_c}\|<\vartheta_*\}$, real on real phases, with $\varphi_c(0)=0$, $D\varphi_c(0)=0$; under the threshold of Section~\ref{app:thresholds} (item 3) $g_0\mathsf C_U\mathcal U_c$ lies in the ball $\|A|_{\mathcal E_c}\|\leq\vartheta_*/2$, so the row solution of Section~\ref{sec:source-phase-primitives} is defined at $A=g_0\mathsf C_UB$ for every $B\in\mathcal U_c$; the determinant $\det(\id-h_U(c)\partial_{A(b_0(c))}\widetilde D_{U,c}(A))$ is positive there, and the following are analytic functions of $B|_{\mathcal E_c\cap E_*}$ on $\mathcal U_c$: $\widetilde D_{U,c}$, the phase $q_U$ on $\mathcal E_c$, the normalised Haar logarithm $\log(\sigma(q_U(\cdot)(b))/\sigma(0))$ for $b\in\mathcal E_c$, the real logarithm of that determinant, and, when $c$ is within $2R_{\mathrm c}$ of $I$, the plaquette terms $s_{p,U}(q_U(\cdot))$ of the plaquettes $p$ with a bond in $\mathcal E_c$ (as functions of $B$ on the intersection of the polydiscs of the rows of $p$, which are rows within $2R_{\mathrm c}+2L$ of $I$); the first four vanish at $B=0$, $\widetilde D_{U,c}$ and $q_U(\cdot)(b_0(c))-g_0(\mathsf C_UB)(b_0(c))$ together with their first derivatives. For every centre $y\in\Omega_1^{(1)}$ within $2R_{\mathrm c}+2L$ of $I$ and every $B\in\mathcal U_y$, the relative means $\mathcal U_{e^{iq}U}(y,x)\mathcal U_U(y,x)^{-1}$, $x\in\mathcal B_y$, at $q=q_U(B)=g_0B$ on $\mathcal E_y$ exist and lie within $r_G$ of $\id$ in operator norm, in the domain of the chart logarithm $\log_G$ of Appendix~\ref{app:background-estimates}, and the gauge term $\mathscr G_{y,U}(q_U(\cdot))$ is an analytic function of $B|_{\mathcal E_y}$ on $\mathcal U_y$ vanishing at $B=0$. Where $(A_{\mathrm e},a)\in\mathcal D_\xi$, these objects coincide with the objects of CRE (1.14)--(1.15), by (B4)(i) and the uniqueness of the small solution (the phase $g_0\mathsf C_U\Psi_\xi(a+A_{\mathrm e})$ lies in the ball of radius $\vartheta_*/2$ on every such row, by the inclusion stated next with $t=1$, $a$ being in the open box by Lemma~\ref{lem:collar-coverage}(iv)). Every $a+t\Psi_\xi(A_{\mathrm e})$ with $a$ in the enlarged box and $t\in[0,1]$ lies in $\mathcal U_c$ and $\mathcal U_y$ for every such row and centre (Lemma~\ref{lem:collar-coverage}(i),(iii): $\Psi_\xi(A_{\mathrm e})=A_{\mathrm e}$ and $|A_{\mathrm e}(b)|<r$ on the bonds of $\mathcal E_c$, $\mathcal E_y$, which are within $2R_{\mathrm c}+4L-1\leq3R_{\mathrm c}$ of $I$, the range of Lemma~\ref{lem:collar-coverage}(i),(iii)), so every near primitive of this subsection is an analytic function of $(a,t)$ there, with its value and its derivatives in $a$ up to order four bounded as in \eqref{eq:c2-primitive-sizes}--\eqref{eq:convex-primitive-derivatives} of Lemma~\ref{lem:c2-primitive-bounds} (under threshold item 4 of Section~\ref{app:thresholds}, $g_0s_\xi^2\leq4g_0(1+r)^3\leq4\omega_*/C_0$, so these are bounds by constants depending on $L$ and the group only).
\end{lemma}

The centred exponent and its primitives are those of CRE (1.14)--(1.16), written as functions of the free vector $B\in\mathfrak g^{\Omega_1^*}$. With $s_{p,U}(q)=1-\operatorname{Re}\tr_N((e^{iq}U)(\partial p))$ for a plaquette $p$ meeting $\Omega_1$ (links off $\Omega_1$ fixed at $V_0$),
\begin{equation}\label{eq:native-source-energy-primitives}
    E_{\xi,p}(B)=-\frac{1}{g_0^2}\bigl[s_{p,U}(q_U(B))-s_{p,U}(0)\bigr]\;,
\end{equation}
and $E_{\xi,y}(B)=-g_0^{-2}\mathscr G_{y,U}(q_U(B))$ from the $y$-summand of \eqref{eq:regular-covariance-gauge}, $y\in\Omega_1$. For either label $\alpha$,
\begin{equation}\label{eq:native-source-energy-residual}
    p_{\xi,\alpha}(B)=E_{\xi,\alpha}(B)-DE_{\xi,\alpha}(0)[B]-\frac{1}{2}D^2E_{\xi,\alpha}(0)[B,B]\;.
\end{equation}
With $\sigma(q)\,\dd q$ the pullback under $q\mapsto e^{iq}$ of the Haar measure $\dd U$ of CRE (1.11) (the density $\sigma$ of (1.14), for the fixed inner product as explained in the conventions; its normalisation enters only the constants $\kappa_\xi$ and $c_\xi$), for every bond $b$ of $\Omega_1$,
\begin{equation}\label{eq:native-source-haar-primitive}
    p^{\mathrm H}_{\xi,b}(B)=\log\sigma(q_U(B)(b))-\log\sigma(0)\;,
\end{equation}
and for every row $c\in\Omega_1^{[1]}$
\begin{equation}\label{eq:native-source-jacobian-primitive}
    p^{\mathrm J}_{\xi,c}(B)
        =\log\Bigl|\det_{\mathfrak g}\left(\id-h_U(c)\partial_{A(b_0(c))}\widetilde D_{U,c}(g_0\mathsf C_UB)\right)\Bigr|\;,
\end{equation}
where $\partial_{A(b_0(c))}\widetilde D_{U,c}$ is the partial derivative of the row solution, regarded as a function of the $|\mathcal E_c|$ independent variables $A(b)$, $b\in\mathcal E_c$, with respect to the variable at $b_0(c)$, evaluated at $A=g_0\mathsf C_UB$ (this is the designated block of CRE's $\frac{\delta}{\delta A}\widetilde D$, proof of Lemma~\ref{lem:primitive-sum}). A primitive is defined at $B$ when the row solutions of its rows are defined at $g_0\mathsf C_UB$ (Section~\ref{sec:source-phase-primitives}: on the domain of integration, on the polydiscs for the rows near $I$, and near $B|_{\text{inputs}}=0$); for a primitive whose row solutions are not defined at $B=0$ on the domain of integration (which happens when $R_1\ne\varnothing$, since the large-fluctuation tests exclude $A=0$ on $R_1$), the Taylor polynomial at $B=0$ in \eqref{eq:native-source-energy-residual} is formed from the germs of Lemma~\ref{lem:row-locality}; CRE's $V$ is the sum of the terms of order higher than two of the expansion at $A=0$ (p.249 L42--44, RG1 p.267), and Lemma~\ref{lem:primitive-sum} identifies the two. The letter $\mathfrak p$ denotes a primitive of any of the three kinds. By Lemma~\ref{lem:row-locality} the free-input set of $E_{\xi,p}$ and $p_{\xi,p}$ is $\bigcup_{r\subset\partial p}\mathcal F_r$, that of $E_{\xi,y}$ and $p_{\xi,y}$ is $\mathcal E_y\cap E_*=\mathcal E_y$, that of $p^{\mathrm H}_{\xi,b}$ is $\mathcal F_b$ and that of $p^{\mathrm J}_{\xi,c}$ is $\mathcal E_c\cap E_*$; the background set of each is the set of bonds whose links enter (bonds of $\Omega_1$ and, for plaquettes and rows at the boundary of $\Omega_1$, bonds of $\Omega_1^c$ carrying $V_0$), as in \eqref{eq:regular-stencil-quadratic-decomposition}; the rows of a primitive are the rows $c$ with $b_0(c)$ in its background set (a gauge primitive $E_{\xi,y}$ has no rows: $\mathcal E_y$ contains no designated bond, and $q_U(B)=g_0B$ on it). A primitive is \emph{near} if its free-input set meets $I$ and \emph{far} otherwise; a far primitive is a function of $A_{\mathrm e}$ and the branch data alone (Lemma~\ref{lem:row-locality}, on the domain of (B4)). A near primitive has a bond $b_1\in I$ in its free-input set, and every bond of $\mathcal E_c$ for a row $c$ of the primitive, and every bond of $\mathcal E_y$ for a gauge primitive, is within $4L+3\leq2R_{\mathrm c}$ of $b_1$ (two bonds of a two-block set are within $2L-1$ of each other, the bonds of a plaquette within $1$); so the rows of a near primitive, and the centre of a near gauge primitive, lie within $2R_{\mathrm c}$ of $I$, and by Lemma~\ref{lem:charts} every near primitive is defined and analytic in $(a,t)$ at the inputs $a+t\Psi_\xi(A_{\mathrm e})$, $a$ in the enlarged box, $t\in[0,1]$. An \emph{arc} $\gamma$ of the contour $C$ is a set of consecutive bonds of $C$ (with their orientations), and $\gamma_+$ is the set of their positive representatives; the free-input and background stencils of an arc are
\begin{equation}\label{eq:native-source-arc-stencils}
    F_\gamma=\bigcup_{r\in\gamma_+}\mathcal F_r,
    \qquad
    D_\gamma=\gamma_+\cup\bigcup_{c\, | \,b_0(c)\in\gamma_+}\mathcal E_c\;.
\end{equation}

\begin{lemma}[exact primitive sum]\label{lem:primitive-sum}
Let $v$ and $V$ be the functions of CRE (1.15), p.249 L42--44: $v(A)$ is the sum of the logarithms of the Haar densities $\sigma(A-h\widetilde D(A))$ over the bonds of $\Omega_1$ and of $\operatorname{Tr}\log(\id-h\frac{\delta}{\delta A}\widetilde D)=\log\det(\id-h\frac{\delta}{\delta A}\widetilde D)$, the logarithm of the Jacobian of (1.14), whose determinant is positive on $\mathcal A_\xi$ ((B4)(iii); factor by factor on the near rows by Lemma~\ref{lem:charts}), and $V(A)$ is the remainder after the Taylor polynomial of degree two at $A=0$ of the action and the gauge-fixing functional at the phase $A-h\widetilde D(A)$ (CRE's ``terms of order higher than two''; no series expansion is used, and the Taylor coefficients at $A=0$ are those of the germs of Lemma~\ref{lem:row-locality}). For $B\in\mathcal A_\xi$, and more generally for every $B\in\mathfrak g^{\Omega_1^*}$ at which every row solution is defined with the row solutions of the near rows those of Lemma~\ref{lem:charts} and those of the far rows the germs at zero argument ($B=0$ on their two-block sets), with $A=g_0\mathsf C_UB$,
\begin{equation}\label{eq:native-source-primitive-sum}
    v(A)-v(0)-g_0^{-2}V(A)
        =\sum_{\alpha}p_{\xi,\alpha}(B)+\sum_{b}p^{\mathrm H}_{\xi,b}(B)+\sum_{c}p^{\mathrm J}_{\xi,c}(B)\;,
\end{equation}
the first sum over the plaquettes $p$ meeting $\Omega_1$ and the centres $y\in\Omega_1^{(1)}$, the second over the bonds $b$ of $\Omega_1$, the third over the rows $c\in\Omega_1^{[1]}$ (the same ranges in every sum $\sum_\alpha$, $\sum_b$, $\sum_c$ below), so that the exponent of the second factor of CRE (1.15) at the free vector $B$, with the quadratic form written through the regional Hessian \eqref{eq:c-regional-hessian},
\begin{equation}\label{eq:native-collar-full-exponent}
    \mathscr E_\xi(B)=-\frac{1}{2}\langle B,H_\xi B\rangle+v(g_0\mathsf C_UB)-g_0^{-2}V(g_0\mathsf C_UB)\;,
\end{equation}
satisfies $\mathscr E_\xi(0)=v(0)=\sum_b\log\sigma(0)$ (sum over the bonds of $\Omega_1$; here $\sigma$ is the density appearing in CRE's $v$, and if CRE's inner density is normalised relative to the outer constant $\sigma_0$ of (1.14)--(1.15) then $v(0)=0$: in either reading $v(0)$ is a constant of the branch depending on $\Omega_1$ only, and $\sigma_0^{|\Omega_1^*|}e^{v(0)}\prod_be^{p^{\mathrm H}_{\xi,b}}$ is CRE's full Haar factor exactly once, the primitives being differences) and
\begin{equation}\label{eq:c-centred-exponent}
    \mathscr E_\xi(B)-\mathscr E_\xi(0)=-\frac{1}{2}\langle B,H_\xi B\rangle+\sum_\alpha p_{\xi,\alpha}(B)+\sum_bp^{\mathrm H}_{\xi,b}(B)+\sum_cp^{\mathrm J}_{\xi,c}(B)
    =\sum_\alpha E_{\xi,\alpha}(B)+\sum_bp^{\mathrm H}_{\xi,b}(B)+\sum_cp^{\mathrm J}_{\xi,c}(B)\;.
\end{equation}
\end{lemma}

\begin{proof}
The sum $\sum_\alpha E_{\xi,\alpha}(B)$ is $-g_0^{-2}$ times the difference of the action plus gauge-fixing functional at the phase $q_U(B)$ and at $q=0$: here the gauge-fixing functional of CRE (1.11), (1.14) is $\mathbf G(\Omega_1^{(1)},e^{iq}U_1)=\mathscr G^{\Omega_1}_{U_1}(q)$ because $U_1$ is in the axial mean gauge (Section~\ref{sec:regular-hessian-stencils}, after \eqref{eq:regular-covariance-hessian}). Since $D\widetilde D_U(0)=0$, $Dq_U(0)[B]=g_0\mathsf C_UB$, and the linear term of the Taylor expansion in $B$ at zero is $\sum_\alpha DE_{\xi,\alpha}(0)[B]=-g_0^{-1}\bigl(DS(U_1)[\mathsf C_UB]+D\mathscr G^{\Omega_1}_U(0)[\mathsf C_UB]\bigr)=0$: the relative gauge functional is a sum of nonnegative terms vanishing at $q=0$, so its first variation vanishes; by CRE (1.12) $U_1$ is a critical point of $S+\mathbf G(\Omega_1^{(1)},\cdot)$ on the configurations with $\mathsf M(U)=V$ on $\Omega_1^{(1)}$ and $U=V_0$ on $\Omega_1^c$, and $\mathbf G(\Omega_1^{(1)},\cdot)\geq0$ attains its minimum $0$ at $U_1$, so $D\mathbf G(\Omega_1^{(1)},U_1)=0$ and $DS(U_1)[\delta]=0$ for every phase $\delta$ supported on the bonds of $\Omega_1$ with $D\mathcal Q_U(0)\delta=LQ_U\delta=0$ (the tangent space of the constraint $\mathsf M(e^{i\delta}U_1)=V$); and $\mathsf C_UB$ is such a phase by $Q_U\mathsf C_U=0$ \eqref{eq:regular-stencil-elimination}. The quadratic term is $-\frac12\langle B,H_\xi B\rangle$ by \eqref{eq:regular-covariance-hessian}, with $A=g_0\mathsf C_UB$: the second variation of the action is $\langle A,\Delta_UA\rangle$ plus the force term $-2\langle J_U,h_U\frac12D^2\mathcal N_U(0)[A,A]\rangle$ from the second-order part $-h_U\widetilde D^{(2)}_U(A)$ of the phase, and that of the gauge functional is $G^{(2),\Omega_1}_U(A)$. Subtracting both from $\sum_\alpha E_{\xi,\alpha}$ gives $\sum_\alpha p_{\xi,\alpha}$, which is therefore $-g_0^{-2}V(A)$. The Haar part of $v$ is $\sum_b\log\sigma(q_U(B)(b))$ bond by bond, so $v$ minus its Haar part at $B=0$ gives $\sum_bp^{\mathrm H}_{\xi,b}$. For the Jacobian, $\frac{\delta}{\delta A}\widetilde D$ has rows indexed by $\Omega_1^{[1]}$ and columns by the bonds of $\Omega_1$, and $h_U$ maps row $c$ to the bond $b_0(c)$; in the ordering (free bonds, designated bonds) the matrix $\id-h_U\frac{\delta}{\delta A}\widetilde D_U$ is block triangular with identity block on the free bonds and designated block $\operatorname{diag}_c(\id-h_U(c)\partial_{A(b_0(c))}\widetilde D_{U,c})$ by Lemma~\ref{lem:two-block} (row $c$ depends on $A|_{\mathcal E_c}$, which contains the single designated bond $b_0(c)$), so its determinant is $\prod_c\det_{\mathfrak g}(\id-h_U(c)\partial_{A(b_0(c))}\widetilde D_{U,c})$, positive on $\mathcal A_\xi$ by (B4)(iii), and at the other points admitted in the statement because each near-row factor is positive by Lemma~\ref{lem:charts} and each far-row factor equals one at zero argument ($D\widetilde D_{U,c}(0)=0$); so $\operatorname{Tr}\log$ of the matrix, CRE's term in $v$, is the real logarithm of the positive determinant, and since each factor is a nonzero real, the logarithm of the absolute value of the product is the sum of the logarithms of the absolute values of the factors, $\sum_c\log|\det_{\mathfrak g}(\cdots)|=\sum_cp^{\mathrm J}_{\xi,c}(B)$; this vanishes at $B=0$ since $D\widetilde D_U(0)=0$; hence $v(0)$ is the Haar part at $B=0$. The quadratic form of (1.15) in the free variable, $\frac12\langle B,C^*\Delta_1CB\rangle$, is $\frac12\langle B,H_\xi B\rangle$ by the identification $\mathsf H^{\Omega_1}_{U_1}=C^*\Delta_1C$ of Section~\ref{sec:regular-hessian-stencils} ($\Delta_1$ of (1.16) already contains $\mathbf G^{(2)}$ and the force term; the $g_0^{-2}$ of the action cancels the $g_0^2$ of $A=g_0\mathsf C_UB$). Subtracting $\mathscr E_\xi(0)=v(0)$ from \eqref{eq:native-collar-full-exponent} and inserting \eqref{eq:native-source-primitive-sum} gives the first form of \eqref{eq:c-centred-exponent}; the second form restores the quadratic term to $\sum_\alpha p_{\xi,\alpha}$.
\end{proof}

Each primitive is assigned a host $Y_\alpha$ in the native localisation family $\mathbf D_0$: lift its free-input and background stencil $\widetilde{\mathcal S}_\alpha$ to $\mathbb Z^4$ about its base (plaquette lower vertex, block centre, initial vertex of its bond, or initial vertex of $b_0(c)$ for $p^{\mathrm J}_{\xi,c}$; the stencil has $\ell^\infty$ diameter at most $4L<M$, below every period, so the lift about the base is well defined), let $Z_\alpha$ be the $M$-cubes $Q_z=Mz+[-M/2,M/2]^4$ meeting it, and put
\begin{equation}\label{eq:part1-potential-host}
    Y_\alpha=\pi_T\Bigl(\bigcup_{z\in\prod_\mu([k^-_{\alpha,\mu}-1,k^+_{\alpha,\mu}+1]\cap\mathbb Z)}Q_z\Bigr),
    \qquad k^-_{\alpha,\mu}=\min_{z\in Z_\alpha}z_\mu,\quad k^+_{\alpha,\mu}=\max_{z\in Z_\alpha}z_\mu\;,
\end{equation}
with $\pi_T$ the reduction modulo the periods; since the stencil has diameter at most $4L<M$, it meets at most two consecutive $M$-cubes in each direction, so the rectangle has side at most $4$ cubes, and every period being at least $8M$, $\pi_T$ is injective on it. $Y_\alpha$ is connected and contains the stencil in its interior. The potential is, for $Y\in\mathbf D_0$ and $a\in\mathfrak g^I$,
\begin{equation}\label{eq:c-potential-definition}
    \mathbf V_\xi(Y,a)=\sum_{\mathfrak p\ \mathrm{near},\ Y_{\mathfrak p}=Y}\mathfrak p(a)\quad\text{if }|a(b)|<r\ \forall b\in E(Y),\qquad\mathbf V_\xi(Y,a)=0\quad\text{otherwise},
\end{equation}
the sum over the near primitives $\mathfrak p$ of all three kinds assigned to $Y$, with $E(Y)=\{b\in I\, | \,b\cap Y\ne\varnothing\}$ and the primitives at zero retained input ($t=0$). It depends on $a|_{E(Y)}$ only, since the free-input set of a primitive assigned to $Y$ lies in $Y$, hence in $E(Y)$ (Lemma~\ref{lem:row-locality}), and it is bounded on the local box by Lemma~\ref{lem:charts} (each summand by $C_{\mathrm p}g_0(1+2r)^3$). On the open box $\{|a(b)|<r\ \forall b\in I\}$, $\sum_Y\mathbf V_\xi(Y,a)=\sum_{\mathfrak p\ \mathrm{near}}\mathfrak p(a)$.

\subsection{The boundary law, the exterior interpolation and the collar}
\label{sec:boundary-law-collar}

\emph{The change of variables.} Let $S_1=(\Omega_1\setminus\Omega_1^{\sim-1})\cup R_1$ (CRE (1.20)) and let $S_1^*$ be its free bonds. CRE (1.22) replaces the variable $A$ of (1.15) on each bond $b\in S_1^*$ by a new variable $A'(b)$ and leaves the other bonds unchanged: $A=\Psi_\xi A'$ with
\begin{equation}\label{eq:native-source-exact-boundary-map}
    (\Psi_\xi A')(b)=\frac{1}{ig_0}\log\left(e^{ig_0A'(b)}e^{-i\omega_b(A'(b))\mathbf H_{1,Ax}(b)}\right)\quad(b\in S_1^*),\qquad
    (\Psi_\xi A')(b)=A'(b)\quad(b\in\Omega_1^*\setminus S_1^*)\;,
\end{equation}
where $\omega_b$ is the $C^\infty$ cutoff $g$ of CRE p.252 L3--4 ($0\leq g\leq1$, $g=0$ on $|A'|\leq\frac43g_0^{-1}\delta_0$, $g=1$ where $|e^{ig_0A'}-1|\geq\frac53\delta_0$) for $b$ in the set $(R_1^{\sim2}\cap S_1)^*$ named in the definition following (1.22) (p.251 L43), and $\omega_b=1$ for the other bonds of $S_1^*$ (p.252 L18 describes the second set as $S_1^*\setminus(R_1^{\sim2})^*$, which differs from the complement of the first set in $S_1^*$ on the bonds joining $R_1^{\sim2}\setminus S_1$ to $S_1\setminus R_1^{\sim2}$; the definition of p.251 L43 is used, and (B4)(ii) refers to the map so defined), $\mathbf H_{1,Ax}(b)$ is the field $\mathbf H_{1,Ax}(\Omega_1\cap\Lambda_1^c,\cdot)$ of CRE (1.21) at the actual branch data, as it appears in (1.22), without the prefactor $L^{-1}$ of (1.21) (a function of $V$ on $\Omega_1\cap\Lambda_1^c$, CRE p.251 L22--25), and $\log$ is the group chart. $\Psi_\xi$ acts bond by bond, and $\psi_b$ denotes its component at $b\in S_1^*$, a $C^\infty$ map ($g$ is $C^\infty$, not analytic) on the open set $\mathcal W_b\subseteq\mathfrak g$ of those $A'(b)$ for which the argument of the chart logarithm lies in the chart domain. By (B4)(ii), $\Psi_\xi$ maps $\mathcal D_\xi$ bijectively onto $\mathcal A_\xi$, the domain of integration of (1.15) with the cutoff (1.25) (Section~\ref{sec:branch-data}), and $D\Psi_\xi$ is invertible there, so that the change of variables formula applies with the Jacobian $|\det D\Psi_\xi|=\prod_{b\in S_1^*}|\det D\psi_b|$; $\chi_{\xi,\mathrm e}(A_{\mathrm e})$ below is set to zero when $A_{\mathrm e}(b)\notin\mathcal W_b$ for some $b\in S_1^*$. Thus $\log|\det D\Psi_\xi(A')|$ depends on $A'|_{S_1^*}$ only, and it is $C^\infty$, which is all that is used of it (Lemma~\ref{lem:collar}: it is a constant on the branch). The integration variable of the branch is $A'=a+A_{\mathrm e}$ with $a\in\mathfrak g^I$ and $A_{\mathrm e}\in\mathfrak g^J$. A bond of $I$ has an endpoint in $\Lambda_1\subseteq\Omega_1^{\sim-2}\cap(R_1'^\sim)^c$, a bond of $S_1^*$ has an endpoint in $S_1$, and $\Omega_1^{\sim-2}$ is separated from $\Omega_1\setminus\Omega_1^{\sim-1}$, as $(R_1'^\sim)^c$ is from $R_1\subseteq R_1'$, by a full layer of $LMR_1$-cubes (the enlargement adjoins every touching cube, Section~\ref{sec:sourced-effective-interaction}); hence, with distances between initial vertices,
\begin{equation}\label{eq:c-collar-distance}
    I\cap S_1^*=\varnothing,\qquad d(I,S_1^*)\geq LMR_1-2\;.
\end{equation}
The filling modification following CRE (1.20) only removes cubes from $\Lambda_1$, so \eqref{eq:c-collar-distance} holds for the final $\Lambda_1$. This gives \eqref{eq:c-bondwise-shift}.

\emph{The cutoff.} The small-field test of CRE (1.18) for a cube $\square'$ of side $L^2M_2R_1$ is the indicator of $\{\sup_{b\in(\square'^{\sim2})^*}|A(b)|<r\}$, $r=\delta_0/g_0$, in CRE's bond norm with the normalisation \eqref{eq:cre-cutoff-normalisation}, and it is the product of one indicator per bond of $(\square'^{\sim2})^*$. Let
\begin{equation}\label{eq:c-good-region}
    \mathcal G_\xi=\bigcup_{\square'\subset R_1^c}(\square'^{\sim2})^*,\qquad R_1^c=\Omega_1^{\sim-1}\setminus R_1\;,
\end{equation}
the union over the good cubes of (1.18). Since $\Lambda_1\subseteq R_1^c$ and both are unions of such cubes, the first two factors of CRE (1.25) are the product of the tests over all good cubes, which by idempotence of indicators is $\prod_{b\in\mathcal G_\xi}\mathbf 1_{\{|A(b)|<r\}}$. The young cutoff is
\begin{equation}\label{eq:native-source-young-cutoff}
    \Theta_\xi(a)=\prod_{b\in I}\mathbf 1_{\{|a(b)|<r\}}\;,
\end{equation}
and the retained cutoff is defined as the remaining factors of (1.25), composed with the change of variables:
\begin{equation}\label{eq:retained-cutoff-composition}
    \chi_{\xi,\mathrm e}(A_{\mathrm e})=\prod_{b\in\mathcal G_\xi\cap J}\mathbf 1_{\{|(\Psi_\xi A_{\mathrm e})(b)|<r\}}\cdot\bigl[\chi^{(0)c}(R_1)\chi_0'(S_1)\bigr](\Psi_\xi A_{\mathrm e})\;,
\end{equation}
where $\chi^{(0)c}(R_1)$ is the product of the large-fluctuation tests of (1.18) over the cubes of $R_1$ and $\chi_0'(S_1)$ the product of the tests of (1.8) over the cubes of $S_1$, both as functions of the variable $A$ of (1.15) on $\Omega_1^*$ (through the free links $e^{ig_0A(b)}U_1(b)$ in the case of (1.8)) and of the branch data ($V_0$ on $\Omega_1^c$, $U_1$ and the local minima $U_{1,\square'}$). CRE's rewriting of these factors in the new variables (p.252 L10--23, (1.23)) is not used; the exact compositions are carried as they stand. Lemma~\ref{lem:collar-coverage}(ii) shows that \eqref{eq:retained-cutoff-composition} is indeed the complement of $\Theta_\xi$ in the exactly transformed cutoff.

\emph{The conditional integrand.} With $\mathscr E_\xi$ the exponent of the second exponential of CRE (1.15), written out in \eqref{eq:native-collar-full-exponent}, the conditional integrand of Section~\ref{sec:sourced-effective-interaction} is
\begin{equation}\label{eq:native-source-actual-functional}
    \mathcal K_\xi(a)=\begin{cases}\exp\Bigl(\mathscr E_\xi\bigl(\Psi_\xi(a+A_{\mathrm e})\bigr)+\log|\det D\Psi_\xi(a+A_{\mathrm e})|\Bigr),&\Theta_\xi(a)=1,\\[2pt] 0,&\Theta_\xi(a)=0,\end{cases}\qquad
    \mathcal J^{\mathrm a}_\xi[f]=\int f(a)\mathcal K_\xi(a)\,\dd a\;,
\end{equation}
the exponent being defined on the box by (B4) and Lemma~\ref{lem:charts}: by Lemma~\ref{lem:collar-coverage}(ii) below, the integrand of CRE (1.15) in the variables of (1.22), with the cutoff (1.25) in its exact composed form, is $\kappa_\xi\,\chi_{\xi,\mathrm e}(A_{\mathrm e})\,\mathcal K_\xi(a)$, as used in \eqref{eq:native-source-physical-ratio} ($\mathscr E_\xi$ and $\kappa_\xi$ are CRE's exponent and prefactor as printed, with $\sigma$, $\sigma_0$ and $z^{(0)}$ for the fixed inner product as explained in the conventions; only the $a$-dependence of $\mathcal K_\xi$ is used below). The change-of-variables term, which CRE expands as $V^{(0)}(S_1)$ ((1.24) and p.252 L33--38), is kept exact inside $\mathcal K_\xi$; Lemma~\ref{lem:collar} shows that it is a constant on the branch.

\emph{The exterior interpolation.} Write $H=H_{\xi,II}$. For $t\in[0,1]$ put
\begin{equation}\label{eq:native-source-boundary-interpolation}
    \begin{aligned}
        P_{\xi,t}(a)&=\sum_{\mathfrak p\ \mathrm{near}}\mathfrak p(a+tA_{\mathrm e})\;,\qquad
        j_\xi(a)=\langle a,H_{\xi,IJ}A_{\mathrm e}\rangle\;,\\
        K_{\xi;t}(a)&=\mathbf 1_{\mathcal D_I}(a)\exp\left(-\frac{1}{2}\langle a,Ha\rangle-tj_\xi(a)+P_{\xi,t}(a)\right)\quad(\mathcal D_I=\{a\, | \,|a(b)|<r\ \forall b\in I\}\text{ the open box, }\mathbf 1_{\mathcal D_I}=\Theta_\xi),\qquad
        I_{\xi;t}=\int K_{\xi;t}(a)\,\dd a,\qquad
        \dd\nu_{\xi;t}=\frac{K_{\xi;t}}{I_{\xi;t}}\,\dd a\;,
    \end{aligned}
\end{equation}
the sum running over the near primitives of Section~\ref{sec:source-phase-primitives} (all three kinds). By Lemma~\ref{lem:charts} and Lemma~\ref{lem:collar}\,(i), each near primitive is defined and analytic in $(a,t)$ on the enlarged box at the inputs $a+tA_{\mathrm e}$, $t\in[0,1]$; $K_{\xi;t}$ (zero off the open box) is therefore bounded and Borel on $\mathfrak g^I$, positive on the open box, and $0<I_{\xi;t}<\infty$. On the open box $P_{\xi,0}(a)=\sum_Y\mathbf V_\xi(Y,a)$ is the centred potential \eqref{eq:c-potential-definition}, and $K_{\xi;0}$ is the centred kernel of Section~\ref{sec:fixed-first-kernel}. This is the interpolation of CRE (3.27)--(3.29) in the exterior field, restricted to the terms that depend on $a$. The Wilson function and the centred coefficient are
\begin{equation}\label{eq:convex-response-operands}
    f_\xi(a)=W_\chi(C;U_\xi(a)),\qquad F_\xi(t)=\int f_\xi\,\dd\nu_{\xi;t}\;,
\end{equation}
with $U_\xi(a)$ the reconstructed fine field \eqref{eq:c-reconstructed-field} at the actual retained data, including the change of variables; it is not interpolated. By Lemmas~\ref{lem:charts} and~\ref{lem:row-locality}, $f_\xi$ is defined and analytic on the enlarged box: each contour link $U_\xi(a)(r)$ depends on $a$ only through $a|_{\mathcal F_r\cap I}$, through a row within $2R_{\mathrm c}$ of $I$ when $\mathcal F_r$ meets $I$, and is constant in $a$ otherwise. Off the enlarged box $f_\xi$ is extended by zero, which affects no integral against $\nu_{\xi;t}$.

\begin{lemma}[the collar]\label{lem:collar}
On every branch:
\begin{enumerate}
\item $\Psi_\xi(A_{\mathrm e})(b)=A_{\mathrm e}(b)$ for every $b\in J$ with $d(b,I)\leq2R_{\mathrm c}$; every near primitive and every entry of $H_{\xi,IJ}$ involves $A_{\mathrm e}$ only on such bonds, so $\mathfrak p(a+t\Psi_\xi(A_{\mathrm e}))=\mathfrak p(a+tA_{\mathrm e})$ for near $\mathfrak p$ and $\langle a,H_{\xi,IJ}\Psi_\xi(A_{\mathrm e})\rangle=j_\xi(a)$. For every row $c$ within $2R_{\mathrm c}$ of $I$ the source phase on the bonds of $\mathcal E_c$ is, componentwise,
\begin{equation}\label{eq:native-collar-source-phase}
    q_U\bigl(\Psi_\xi(a+A_{\mathrm e})\bigr)(b)=q_U(a+A_{\mathrm e})(b)=\bigl[g_0\mathsf C_U(a+A_{\mathrm e})-h_U\widetilde D_U\bigl(g_0\mathsf C_U(a+A_{\mathrm e})\bigr)\bigr](b)\qquad(b\in\mathcal E_c)\;.
\end{equation}
\item With the branch constant
\begin{equation}\label{eq:native-collar-boundary-constant}
    c_\xi=\mathscr E_\xi(0)-\frac12\langle\Psi_\xi(A_{\mathrm e}),H_\xi\Psi_\xi(A_{\mathrm e})\rangle+\sum_{\mathfrak p\ \mathrm{far}}\mathfrak p\bigl(\Psi_\xi(A_{\mathrm e})\bigr)+\log|\det D\Psi_\xi(A_{\mathrm e})|\;,
\end{equation}
where $\mathscr E_\xi(0)=v(0)$ and the far primitives are evaluated at the retained field alone,
\begin{equation}\label{eq:c-collar-identity}
    \mathcal K_\xi(a)=e^{c_\xi}K_{\xi;1}(a)\quad(a\in\mathfrak g^I),\qquad
    \mathcal J^{\mathrm a}_\xi[f]=e^{c_\xi}I_{\xi;1}\int f\,\dd\nu_{\xi;1},\qquad
    \frac{\mathcal J^{\mathrm a}_\xi[f_\xi]}{\mathcal J^{\mathrm a}_\xi[1]}=F_\xi(1)\;.
\end{equation}
\end{enumerate}
\end{lemma}

\begin{proof}
(i) By Lemma~\ref{lem:collar-coverage}(i), a bond within $2R_{\mathrm c}$ of $I$ is not in $S_1^*$, where $\Psi_\xi$ is the identity. A near primitive has all its inputs within $4L+3\leq2R_{\mathrm c}$ of $I$ (Section~\ref{sec:source-phase-primitives}); an entry $(H_{\xi,IJ})_{bb'}$ with $b\in I$ vanishes unless $d(b,b')\leq\rho_L=4L<2R_{\mathrm c}$ (Lemma~\ref{lem:hessian-gauge-locality}), since $R_{\mathrm c}\geq2R_{\mathrm{loc}}+1=4L+5$. The phase on a row $c$ within $2R_{\mathrm c}$ of $I$ depends on the free vector on $\mathcal E_c\cap E_*$ (Lemma~\ref{lem:row-locality}), and every bond of $\mathcal E_c$ is within $2R_{\mathrm c}$ of $I$ by the definition of the row distance (Section~\ref{sec:source-phase-primitives}), so $\Psi_\xi$ is the identity on these bonds; on $\mathcal E_c$ only the row $c$ contributes to $h_U\widetilde D_U$ (Lemma~\ref{lem:row-locality}), which gives \eqref{eq:native-collar-source-phase} as an identity between the $\mathcal E_c$-components.

(ii) Put $A^\Psi_{\mathrm e}=\Psi_\xi(A_{\mathrm e})\in\mathfrak g^J$, so that $\Psi_\xi(a+A_{\mathrm e})=a+A^\Psi_{\mathrm e}$ by \eqref{eq:c-bondwise-shift}. By Lemma~\ref{lem:primitive-sum},
\begin{equation*}
    \mathscr E_\xi(a+A^\Psi_{\mathrm e})=\mathscr E_\xi(0)-\frac12\langle a,Ha\rangle-\langle a,H_{\xi,IJ}A^\Psi_{\mathrm e}\rangle-\frac12\langle A^\Psi_{\mathrm e},H_\xi A^\Psi_{\mathrm e}\rangle+\sum_{\mathfrak p\ \mathrm{near}}\mathfrak p(a+A^\Psi_{\mathrm e})+\sum_{\mathfrak p\ \mathrm{far}}\mathfrak p(A^\Psi_{\mathrm e})\;,
\end{equation*}
where the far primitives do not depend on $a$ because their free-input sets miss $I$. By (i) the third term is $-j_\xi(a)$ and the near sum is $P_{\xi,1}(a)$. The Jacobian factor depends on $(a+A_{\mathrm e})|_{S_1^*}=A_{\mathrm e}|_{S_1^*}$ only. The constant $c_\xi$ is finite: $\det D\Psi_\xi(A_{\mathrm e})\neq0$ by (B4)(ii) (the differential of (1.22) is invertible on $\mathcal D_\xi$, and $D\Psi_\xi$ at $a+A_{\mathrm e}$ depends on $A_{\mathrm e}|_{S_1^*}$ only), and the far primitives are finite: their free-input sets miss $I$, so their values at $A^\Psi_{\mathrm e}$ are their values at $a_0+A^\Psi_{\mathrm e}=\Psi_\xi(a_0+A_{\mathrm e})\in\mathcal A_\xi$ for any $a_0$ with $(A_{\mathrm e},a_0)\in\mathcal D_\xi$ (a branch has one), where every row solution is defined and the primitives are finite by (B4). Collecting the $a$-independent terms gives $\mathcal K_\xi=e^{c_\xi}K_{\xi;1}$, and the two remaining identities follow by integration, dividing by $\mathcal J^{\mathrm a}_\xi[1]=e^{c_\xi}I_{\xi;1}>0$.
\end{proof}

The constant $c_\xi$ depends on the branch data (through $A_{\mathrm e}$, $U_1$, $V_0$ and $\Omega_1$) and is not a universal constant: it enters the outer weight $\mathcal J^{\mathrm a}_\xi[1]=e^{c_\xi}I_{\xi;1}$ of the branch in \eqref{eq:native-source-physical-ratio}. It is never estimated separately: Section~\ref{sec:one-step-insertion} compares $\mathcal O[\mathcal J^{\mathrm a}_\xi[1]\,F_\xi(1)]$ with $\mathcal O[\mathcal J^{\mathrm a}_\xi[1]]$, the same nonnegative weight on both sides.

\begin{lemma}[boundary response]\label{lem:boundary-response}
For bounded Borel $f$ on $\mathfrak g^I$, with $\operatorname{Cov}_\nu(f,l)=\int fl\,\dd\nu-\int f\,\dd\nu\int l\,\dd\nu$,
\begin{equation}\label{eq:fixed-probe-boundary-functional}
    \int f\,\dd\nu_{\xi;1}=\int f\,\dd\nu_{\xi;0}+\mathcal B_\xi[f],\qquad
    \mathcal B_\xi[f]=\int_0^1\operatorname{Cov}_{\nu_{\xi;t}}(f,-j_\xi+\partial_tP_{\xi,t})\,\dd t\;,
\end{equation}
where $\partial_tP_{\xi,t}(a)=\sum_{\mathfrak p\ \mathrm{near}}D\mathfrak p(a+tA_{\mathrm e})[A_{\mathrm e}]$.
\end{lemma}

\begin{proof}
For $a$ in the box the exponent $E_t(a)=-\frac12\langle a,Ha\rangle-tj_\xi(a)+P_{\xi,t}(a)$ is jointly analytic in $(a,t)$ on a neighbourhood of $\overline{\mathcal D_I}\times[0,1]$, since each near primitive is analytic in the free vector on its row polydiscs (Lemma~\ref{lem:charts}) and the input $a+tA_{\mathrm e}$ is affine in $(a,t)$ with values in those polydiscs; its $t$-derivative $-j_\xi(a)+\partial_tP_{\xi,t}(a)$, with $\partial_tP_{\xi,t}(a)=\sum_{\mathfrak p}D\mathfrak p(a+tA_{\mathrm e})[A_{\mathrm e}]$ the derivative in the direction $A_{\mathrm e}$, is continuous and hence bounded on $\overline{\mathcal D_I}\times[0,1]$. Hence $t\mapsto\int fK_{\xi;t}\,\dd a$ and $t\mapsto I_{\xi;t}$ are $C^1$ with derivatives $\int f\,\partial_tE_t\,K_{\xi;t}\,\dd a$ and $\int\partial_tE_t\,K_{\xi;t}\,\dd a$ (differentiation under the integral over the bounded box, dominated by a constant times $K_{\xi;t}\leq e^{\sup|E_t|}$), so $F(t)=\int f\,\dd\nu_{\xi;t}$ has $F'(t)=\operatorname{Cov}_{\nu_{\xi;t}}(f,\partial_tE_t)$. Integrate over $[0,1]$.
\end{proof}

By Lemmas~\ref{lem:collar} and~\ref{lem:boundary-response}, on every branch
\begin{equation}\label{eq:c-exterior-response}
    \frac{\mathcal J^{\mathrm a}_\xi[f_\xi]}{\mathcal J^{\mathrm a}_\xi[1]}=F_\xi(1)=F_\xi(0)+\mathcal B_\xi[f_\xi]\;,
\end{equation}
the first term being the centred expectation of Section~\ref{sec:probe-kernels} and the second the exterior response, estimated in Appendix~\ref{app:exterior-chessboard}.

\begin{lemma}[collar coverage]\label{lem:collar-coverage}
Let $R_{\mathrm{loc}}=2L+2$, which exceeds the $\ell^\infty$ diameter $2L-1$ of every two-block set $\mathcal E_c$, hence of the sets $\mathcal E_y$, $\mathcal F_r$, $\mathcal A_r$ and of the free-input and background sets of the block, row, Haar and Jacobian primitives (those of a plaquette primitive have diameter at most $4L<2R_{\mathrm{loc}}$, proof of Lemma~\ref{lem:hessian-gauge-locality}), and let $R_{\mathrm c}\geq2R_{\mathrm{loc}}+1$ be the collar range fixed in Appendix~\ref{app:background-estimates} after \eqref{eq:regular-covariance-radius-replay}; it depends on $L$ and the group only. Under \eqref{eq:c-standing-conditions}, which include $M\geq8LM_2$ and $LMR_1>4R_{\mathrm c}+4L^2M_2R_1+8$, on every branch:
\begin{enumerate}
\item Every site within $\ell^\infty$ distance $LMR_1-1$ of $\Lambda_1$ lies in $\Omega_1^{\sim-1}\setminus R_1'\subseteq R_1^c$; $I\subseteq\mathcal G_\xi\subseteq\Omega_1^*$; and every free bond $b$ of $\Omega_1$ with $d(b,I)\leq3R_{\mathrm c}$ lies in $\mathcal G_\xi\setminus S_1^*$.
\item The cutoff (1.25) composed with the change of variables, as a function of $A'=a+A_{\mathrm e}$, equals $\Theta_\xi(a)\chi_{\xi,\mathrm e}(A_{\mathrm e})$ with $\chi_{\xi,\mathrm e}$ of \eqref{eq:retained-cutoff-composition}, and $\chi_{\xi,\mathrm e}$ depends on $A_{\mathrm e}$ and the branch data only.
\item $|A_{\mathrm e}(b)|<r$ for every $b\in J$ with $d(b,I)\leq3R_{\mathrm c}$; that is,
\begin{equation}\label{eq:convex-native-radius}
    r_{\mathrm e,\xi}=\max\bigl(\{|A_{\mathrm e}(b)|\, | \,b\in J,\ d(b,I)\leq3R_{\mathrm c}\}\cup\{0\}\bigr)<r\;.
\end{equation}
\item For an admissible pair and a branch value of $A_{\mathrm e}$, $\{a\in\mathfrak g^I:(A_{\mathrm e},a)\in\mathcal D_\xi\}=\{a:\Theta_\xi(a)=1\}$, the open box of \eqref{eq:native-source-young-cutoff}.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) $\Omega_1$ and $R_1'$ are unions of $LMR_1$-cubes (CRE (1.10), p.251 L3) and $\Lambda_1\subseteq\Omega_1^{\sim-2}\cap(R_1'^\sim)^c$, so every site within $\ell^\infty$ distance $LMR_1-1$ of $\Lambda_1$ lies in $\Omega_1^{\sim-1}\setminus R_1'\subseteq R_1^c$. Let $b$ be a free bond of $\Omega_1$ with $d(b,I)\leq3R_{\mathrm c}$: its initial vertex is within $3R_{\mathrm c}+1$ of a site of $\Lambda_1$, so the cube $\square'$ of side $L^2M_2R_1$ containing it lies within $3R_{\mathrm c}+1+L^2M_2R_1<LMR_1-1$ of $\Lambda_1$ (\eqref{eq:c-standing-conditions}), hence $\square'\subseteq R_1^c$ is a good cube; $b$ has an endpoint in $\square'\subseteq\square'^{\sim2}$ and is not designated, so $b\in(\square'^{\sim2})^*\subseteq\mathcal G_\xi$. By \eqref{eq:c-collar-distance}, $d(I,S_1^*)\geq LMR_1-2>3R_{\mathrm c}$, so $b\notin S_1^*$. Taking $d(b,I)=0$ gives $I\subseteq\mathcal G_\xi$ (a bond of $\Lambda_1$ is a bond of $\Omega_1$, and a designated bond with an endpoint in $\Lambda_1$ belongs to a row of $\Lambda_1$ because $\Lambda_1$ is a union of blocks, so $I\subseteq\Omega_1^*$). For $\mathcal G_\xi\subseteq\Omega_1^*$: a good cube $\square'\subseteq\Omega_1^{\sim-1}$ has $\square'^{\sim2}\subseteq\Omega_1$ since $2L^2M_2R_1<LMR_1$ ($M\geq8LM_2$); a bond of $(\square'^{\sim2})^*$ has an endpoint in $\Omega_1$, and a designated bond with an endpoint in $\square'^{\sim2}$, a union of blocks, belongs to a row with an endpoint in $(\square'^{\sim2})^{(1)}$ and is excluded by the definition of $^*$ in (1.8).

(ii) The first two factors of (1.25) are $\prod_{b\in\mathcal G_\xi}\mathbf 1_{\{|A(b)|<r\}}$ at $A=\Psi_\xi(a+A_{\mathrm e})$. For $b\in I\subseteq\mathcal G_\xi$, $A(b)=a(b)$ by \eqref{eq:c-bondwise-shift}, giving $\Theta_\xi(a)$; for $b\in\mathcal G_\xi\setminus I=\mathcal G_\xi\cap J$ (by (i)), $A(b)=(\Psi_\xi A_{\mathrm e})(b)$, giving the first factor of \eqref{eq:retained-cutoff-composition}. The factor $\chi^{(0)c}(R_1)$ involves $A$ on $(\square'^{\sim2})^*$ for cubes $\square'\subseteq R_1$, and $\chi_0'(S_1)$ involves the free links $e^{ig_0A(b)}U_1(b)$ on bonds $b\in(\square'^{\sim2})^*\cap\Omega_1^*$ for cubes $\square'\subseteq S_1$, together with $V_0$ on the bonds of $(\square'^{\sim2})^*$ with both endpoints off $\Omega_1$, $U_1$ and $U_{1,\square'}$. All these bonds have both endpoints within $2L^2M_2R_1+1$ of $R_1\cup S_1$, while every site of $\Lambda_1$ is at distance at least $LMR_1>2L^2M_2R_1+1$ from $R_1\cup S_1$; so none of them has an endpoint in $\Lambda_1$, none lies in $I$, and on them $A=\Psi_\xi(A_{\mathrm e})$. Hence the last two factors of (1.25), composed with $\Psi_\xi$, are functions of $\Psi_\xi(A_{\mathrm e})$ and the branch data.

(iii) For $b\in J$ with $d(b,I)\leq3R_{\mathrm c}$, (i) gives $b\in\mathcal G_\xi\cap J$ and $b\notin S_1^*$, so $\mathbf 1_{\{|(\Psi_\xi A_{\mathrm e})(b)|<r\}}=\mathbf 1_{\{|A_{\mathrm e}(b)|<r\}}$ is a factor of $\chi_{\xi,\mathrm e}(A_{\mathrm e})$, which is nonzero on a branch (the maximum over the empty set is read as $0$).

(iv) Let $a_0$ be such that $(A_{\mathrm e},a_0)\in\mathcal D_\xi$ (a branch), so $B_0=\Psi_\xi(a_0+A_{\mathrm e})\in\mathcal A_\xi$ and, by (ii), $\Theta_\xi(a_0)\chi_{\xi,\mathrm e}(A_{\mathrm e})\neq0$. Let $a$ be any point of the open box and $B=\Psi_\xi(a+A_{\mathrm e})=a+\Psi_\xi(A_{\mathrm e})$ (\eqref{eq:c-bondwise-shift}; $\Psi_\xi(A_{\mathrm e})$ is defined, since $B_0$ is). It is to be shown that $B\in\mathcal A_\xi$, by checking the three conditions of \eqref{eq:c-free-variable-domain}. The cutoff (1.25) at $B$ is $\Theta_\xi(a)\chi_{\xi,\mathrm e}(A_{\mathrm e})=\chi_{\xi,\mathrm e}(A_{\mathrm e})\neq0$, by (ii). Every row equation has a small solution at $g_0\mathsf C_UB$: on the near rows by Lemma~\ref{lem:charts}, since $B=a+A_{\mathrm e}$ on their two-block sets by (i) and lies in the row polydiscs by (iii); on the far rows the argument $(g_0\mathsf C_UB)|_{\mathcal E_c}$ equals $(g_0\mathsf C_UB_0)|_{\mathcal E_c}$ (Lemma~\ref{lem:row-locality}: it involves $B$ only through $B|_{\mathcal E_c\cap E_*}$, and $\mathcal E_c$ misses $I$ for a far row, every bond of which is at distance more than $2R_{\mathrm c}+2L-(2L-1)>0$ from $I$), where the small solution exists because $B_0\in\mathcal A_\xi$. So $U(B)$ is defined, and on the far rows and the bonds of $\Omega_1^c$ its links are those of $U(B_0)$. It remains to show $U(B)\in D^{(11)}(V,V_0)$. The restriction to $\Omega_1^c$ is $V_0$. The block averages are defined on every row of $\Omega_1$ and satisfy the constraint $\bar U=V$ there: $U(B)=e^{iq_U(B)}U_1$ with $q_U(B)=A-h_U\widetilde D_U(A)$, $A=g_0\mathsf C_UB$, and on every row $c$ the row equation gives $\mathcal Q_{U,c}(q_U(B))=LQ_Uq_U(B)+\mathcal N_{U,c}(q_U(B))=LQ_U(A-h_U\widetilde D_U(A))+\widetilde D_{U,c}(A)=-\widetilde D_{U,c}(A)+\widetilde D_{U,c}(A)=0$, using $Q_U\mathsf C_U=0$ and $LQ_Uh_U=\id$; so $\mathsf M(U(B))(c)=\mathsf M(U_1)(c)=V(c)$ by \eqref{eq:native-source-constraint-normalisation} and (B1). The constraint $\bar V_0=V$ on the rows of $\Omega_1^c$ involves $V_0$ and $V$ only and holds because the pair is admissible. The tests of (1.11): $\chi_1(\Omega_1)$ is a function of $V$, $\zeta(\Omega_1^c)$ does not depend on $a$ (\eqref{eq:c-large-field-factor}), and both are nonzero at $U(B_0)\in D^{(11)}(V,V_0)$. The factor $\chi_0'(\Omega_1)$ is the product of $\chi_0'(\square')$ over the $L^2M_2R_1$-cubes $\square'\subseteq\Omega_1$ (CRE (1.8)); the cubes of $S_1=(\Omega_1\setminus\Omega_1^{\sim-1})\cup R_1$ (CRE (1.20)) give the factor $\chi_0'(S_1)$ of (1.25), nonzero at $B$ by the first condition, and every other cube $\square'\subseteq\Omega_1$ lies in $\Omega_1^{\sim-1}\setminus R_1$, so it is a good cube with $\square'^{\sim4}\subseteq\Omega_1$ (as $4L^2M_2R_1<LMR_1$), and its factor $\chi_0'(\square')$ is the second factor of (1.19), equal to one at $B$ by (B3)(ii), the retained factors being nonzero there. Hence $U(B)\in D^{(11)}(V,V_0)$, so $B\in\mathcal A_\xi$ and $(A_{\mathrm e},a)\in\mathcal D_\xi$. Conversely $(A_{\mathrm e},a)\in\mathcal D_\xi$ gives $\Psi_\xi(a+A_{\mathrm e})\in\mathcal A_\xi$, so the cutoff (1.25) is nonzero there, and $\Theta_\xi(a)=1$ by (ii).
\end{proof}

The exterior response has Ba{\l}aban's integration-by-parts form (CRE (3.28)). Put
\begin{equation}\label{eq:native-collar-mean-direction}
    \ell_\xi=H_{\xi,IJ}A_{\mathrm e},\qquad m_\xi=H_{\xi,II}^{-1}\ell_\xi,\qquad D_{m_\xi}f(a)=\frac{\dd}{\dd s}f(a+sm_\xi)\Big|_{s=0}\;,
\end{equation}
let $\mathcal D_I$ be the open box of \eqref{eq:native-source-boundary-interpolation}, with outward unit normal $n(a)$ and surface measure $\dd S$ for the Euclidean structure of $\langle\cdot,\cdot\rangle$ on $\mathfrak g^I$ (for which $\dd a$ is Lebesgue measure), defined almost everywhere on its Lipschitz boundary (a product of balls of the norm $|\cdot|$), and
\begin{equation}\label{eq:native-collar-flux}
    \mathsf Q_{\xi,t}(a)=-\frac{1}{2}\langle a,H_{\xi,II}a\rangle-t\langle\ell_\xi,a\rangle+P_{\xi,t}(a),\qquad
    \mathcal T_{\xi,t}[f]=\frac{1}{I_{\xi;t}}\int_{\partial\mathcal D_I}f(a)e^{\mathsf Q_{\xi,t}(a)}\langle m_\xi,n(a)\rangle\,\dd S(a)\;,
\end{equation}
so that $K_{\xi;t}=\mathbf 1_{\mathcal D_I}e^{\mathsf Q_{\xi,t}}$ and $j_\xi(a)=\langle\ell_\xi,a\rangle$.

\begin{lemma}[integration by parts with the hard cutoff]\label{lem:ibp}
Assume Lemma~\ref{lem:regular-bounds}, so that $H_{\xi,II}$ is invertible and $m_\xi$ is defined. If $f$ is $C^1$ on a neighbourhood of $\overline{\mathcal D_I}$, then for every $t\in[0,1]$, with expectations under $\nu_{\xi;t}$,
\begin{equation}\label{eq:native-collar-source-ibp}
    \operatorname{Cov}(f,-j_\xi+\partial_tP_{\xi,t})
        =-\mathbb E[D_{m_\xi}f]+\operatorname{Cov}(f,\partial_tP_{\xi,t}-D_{m_\xi}P_{\xi,t})+\mathcal T_{\xi,t}[f]-\mathbb E[f]\mathcal T_{\xi,t}[1]\;.
\end{equation}
\end{lemma}

\begin{proof}
$e^{\mathsf Q_{\xi,t}}$ is $C^1$ on the enlarged box, a neighbourhood of $\overline{\mathcal D_I}$, by Lemma~\ref{lem:charts}. The divergence theorem on the bounded Lipschitz domain $\mathcal D_I$ gives $\int_{\mathcal D_I}D_{m_\xi}(fe^{\mathsf Q_{\xi,t}})\,\dd a=I_{\xi;t}\mathcal T_{\xi,t}[f]$, and $D_{m_\xi}\mathsf Q_{\xi,t}=-j_\xi-t\langle m_\xi,\ell_\xi\rangle+D_{m_\xi}P_{\xi,t}$ because $\langle m_\xi,H_{\xi,II}a\rangle=\langle\ell_\xi,a\rangle=j_\xi(a)$. Dividing by $I_{\xi;t}$, $\mathbb E[D_{m_\xi}f]+\mathbb E[f(-j_\xi-t\langle m_\xi,\ell_\xi\rangle+D_{m_\xi}P_{\xi,t})]=\mathcal T_{\xi,t}[f]$. Subtract the same identity for $f=1$ multiplied by $\mathbb E[f]$; the constant $-t\langle m_\xi,\ell_\xi\rangle$ cancels, and the two covariances combine as stated.
\end{proof}

Since $f_\xi$ is analytic on the enlarged box (after \eqref{eq:convex-response-operands}), Lemma~\ref{lem:ibp} applies to $f_\xi$. The estimate of the right-hand side is Lemma~\ref{lem:exterior-response-bound} in Appendix~\ref{app:exterior-chessboard}.

\subsection{A fixed walk expansion of the centred kernel}
\label{sec:fixed-first-kernel}

Write $H=H_{\xi,II}$ and $\mathsf G=H^{-1}$. Notation of Sections~\ref{sec:fixed-first-kernel}--\ref{sec:joint-extraction}: $E_0=E(Z_0)$ denotes the seed bonds (the designated bonds $E_0$ of \eqref{eq:constraint-extension} do not occur here); $N_i(Y)$ is a cube count (the numerator $N_1(V)$ of \eqref{eq:first-step} does not occur); $\sigma_\Delta$ are weakening variables (the Haar density $\sigma$ does not occur); $B$ is the conditioned variable on $\mathfrak g^{E_0}$ (the free vector $B$ of Sections~\ref{sec:regular-hessian-stencils}--\ref{sec:boundary-law-collar} does not occur); $D$ without argument is a component; $J_D$, $J_Z$ are sets of young bonds (the retained carrier $J$ does not occur); $\delta_M$ is a decay constant (the Mayer constant $\delta$ of Appendix~\ref{app:background-estimates} does not occur); $\gamma=\gamma_L/2$ is a spectral bound (arcs $\gamma$ of the contour do not occur); a connected final domain is any nonempty wall-connected union of $LM$-cubes; and $a_j$, $a_*$, $a_W$ are coefficients and weights, distinct from the young variable $a$. By Lemma~\ref{lem:regular-bounds}, on every branch
\begin{equation}\label{eq:fixed-kernel-inputs}
    \gamma\id\leq H\leq h_+\id,\qquad\gamma=\frac{\gamma_L}{2},\quad h_+=m_L\;,
\end{equation}
and by Lemma~\ref{lem:hessian-gauge-locality}, whose entries $\mathsf H^{\Omega_1}_{U_1}(b,c)$ at $b,c\in I$ are those of $H$, $H(b,c)=0$ when $d_{\infty,T}(b_-,c_-)>R$, where $R\geq\max(\rho_L,2)$ is fixed so that $H(b,c)$ is determined by the background, by the region $\Omega_1$ and by membership in $I$ within distance $R$ of $b_-$. Norms on $\mathfrak g^I$ are those of the invariant real inner product, with Hermitian extension on the complexification. The constants $\gamma,h_+,R$ depend on $L$ and the group only (with the invariant inner product fixed in the conventions), and every choice below is made before the branch fields and the coupling, in the order $\gamma,h_+,R$; then $\kappa_1$; then $n_*,R_*,C_{\mathrm w}$; then $M_1$, $\rho_{\mathrm w}=R_*+R+1+2M_1$ and $M$; then $g_0$. None of $\gamma,h_+,R,\kappa_1,C_{\mathrm d}$ depends on $M_1$, $M$ or the coupling ($C_{\mathrm d}$ of Lemma~\ref{lem:gaussian-payment} is a function of $\gamma,h_+,d_G$).

Fix an aligned auxiliary cubical partition of side $M_1$ in fine units, with cubes $M_1z+[-M_1/2,M_1/2]^4$; the $M$-cube partition $\pi_0$ and the $LM$-cube partition $\pi_1$ are nested over it, $M/M_1$ and $L$ are odd integers, every fine period is a multiple of $LMR_1$ and at least $8M$, and $M_1$ and $M$ are odd (\eqref{eq:c-standing-conditions}: $M_1$, $M_2$, $M$ are powers of the odd number $L$), so that cube faces lie at half-integer coordinates and no lattice site lies on a face. Cubes and neighbourhoods are periodic images; all radii used below are smaller than $M$, so the periods $\geq8M$ make every neighbourhood a lifted copy of its $\mathbb Z^4$ model. Set $s_H=2h_+$, $\mathsf A_H=\id-H/s_H$, $q_H=1-\gamma/s_H<1$ and $V_n=4(2nR+3)^4$. A block of $\mathsf A_H^n$ vanishes beyond distance $nR$, has operator norm at most $q_H^n$ and depends only on coefficients within distance $(n+1)R$ of its first bond, so its block-norm row and column sums are at most $V_nq_H^n$. Let $\kappa_1\geq1$ be the constant fixed in Appendix~\ref{app:background-estimates}, Section~\ref{app:mayer-constants} (\eqref{eq:c2-mayer-constants}; it depends on $L$ and the group only), the spatial Cauchy radius being $e^{\kappa_1}$; put $a_*=e^{81\kappa_1}$ and choose $n_*$ with
\begin{equation}\label{eq:fixed-kernel-grouping}
    \beta=V_{n_*}q_H^{n_*},\qquad a_*\beta\leq\frac{1}{4}\;.
\end{equation}
Put $R_*=n_*R$, $\mathsf P_H=\mathsf A_H^{n_*}$, $V_*=V_{n_*}$ and $a_j=4^{-j}\binom{2j}{j}\in(0,1]$. The binomial series gives
\begin{equation}\label{eq:fixed-kernel-square-root-series}
    \mathsf G^{1/2}=s_H^{-1/2}\sum_{r=0}^{n_*-1}\sum_{m=0}^\infty a_{n_*m+r}\mathsf A_H^r\mathsf P_H^m\;.
\end{equation}
For a free bond $b$ let $D_b$ be the union of the $M_1$-cubes meeting the closed ball of radius $R_*+R+1$ about $b_-$, with one further $M_1$-layer; it is connected, lies in the ball of radius $\rho_{\mathrm w}=R_*+R+1+2M_1$ about $b_-$, and contains the bonds and coefficient stencils of every block starting at $b$ in \eqref{eq:fixed-kernel-square-root-series} with an interior margin. Choose
\begin{equation}\label{eq:fixed-kernel-collar-size}
    M\geq4(\rho_{\mathrm w}+R+R_*+2)\;.
\end{equation}
Then $2\rho_{\mathrm w}<M\leq LM$, so $D_b$ meets at most $2^4=16\leq81$ cubes of $\pi_1$. A walk is a tuple $\omega=(r;b_0,\ldots,b_{m+1})$ with nonzero block product
\begin{equation}\label{eq:fixed-kernel-walk-block}
    (\mathsf A_H^r)_{b_0b_1}\prod_{j=1}^m(\mathsf P_H)_{b_jb_{j+1}}\;,
\end{equation}
and localisation domain $D(\omega)=\bigcup_{j\leq m}D_{b_j}$, which is connected and contains both endpoint bonds in its interior.

For an $M$-cube $Q$ let $Q^\sim$ be the union of the $3^4$ cubes of $\pi_0$ meeting $Q$, so that $\operatorname{int}Q^\sim$ is the open $\ell^\infty$ neighbourhood of radius $M$ of $Q$. For a union $Z_0$ of $M$-cubes (the seed, selected in Section~\ref{sec:selected-integrals}) let $\widetilde Z_0=\bigcup_{Q\subseteq Z_0}Q^\sim$ be its one-$M$-layer enlargement and $Z_0'$ the union of the $LM$-cubes meeting $\operatorname{int}Q^\sim$ for some cube $Q$ of $Z_0$. Both are unions over the cubes of $Z_0$, so both commute with unions of seeds, and $\operatorname{int}Q^\sim\subseteq\operatorname{int}Z_0'$ for every cube $Q$ of $Z_0$, since every $LM$-cube incident to a point of the open set $\operatorname{int}Q^\sim$ meets it. Assign $\sigma_\Delta=1$ to the cubes $\Delta\in\pi_1$ inside $Z_0'$ and complex variables $\sigma_\Delta$ to the others, and weight each walk by
\begin{equation}\label{eq:fixed-kernel-walk-weakening}
    w_\omega(\sigma)=\prod_{\Delta\in\pi_1,\ \Delta\not\subseteq Z_0',\ \Delta\cap D(\omega)\ne\varnothing}\sigma_\Delta\;.
\end{equation}
The weighted series is
\begin{equation}\label{eq:c-weighted-walk-series}
    S_{Z_0}(\sigma)_{b_0b_{m+1}}=s_H^{-1/2}\sum_{r=0}^{n_*-1}\sum_{m=0}^\infty a_{n_*m+r}\sum_{b_1,\ldots,b_m}w_\omega(\sigma)\,(\mathsf A_H^r)_{b_0b_1}\prod_{j=1}^m(\mathsf P_H)_{b_jb_{j+1}},\qquad\omega=(r;b_0,\ldots,b_{m+1})\;,
\end{equation}
the inner sum running over the intermediate bonds; $S_{Z_0}(1)=\mathsf G^{1/2}$ by \eqref{eq:fixed-kernel-square-root-series}.

\begin{lemma}[the walk series]\label{lem:walk-series}
On $|\sigma_\Delta|\leq e^{\kappa_1}$, $|w_\omega(\sigma)|\leq a_*^{m+1}$, and the series $S_{Z_0}(\sigma)$ converges absolutely in both Schur norms, uniformly on the polydisc, with operator norm and absolute block row and column sums at most
\begin{equation}\label{eq:fixed-kernel-schur-bound}
    C_{\mathrm w}=s_H^{-1/2}\frac{n_*V_*a_*}{1-a_*\beta}\;.
\end{equation}
It is holomorphic on the polydisc of radii $e^{\kappa_1+(\log2)/81}$. The bound is independent of $Z_0$, the volume and the number of its components. Under a site gauge transformation $u$ of the background, $S_{Z_0}(\sigma)$ formed from $\iota_I^*\mathsf H^{\Omega_1}_{U_1^u}\iota_I$ in place of $H$ equals $R_{u,*}S_{Z_0}(\sigma)R_{u,*}^{-1}$, with $R_{u,*}$ restricted to $I$ and the same domains and weights.
\end{lemma}

\begin{proof}
$D(\omega)$ meets at most $81(m+1)$ cubes of $\pi_1$, which gives the weight bound. The unweakened grouped products have absolute block row and column sums at most $V_*\beta^m$ after summing the intermediate bonds, so the weighted series is dominated by $s_H^{-1/2}n_*V_*\sum_m(a_*\beta)^ma_*$. Increasing each radius by the factor $e^{(\log2)/81}$ multiplies $a_*$ by $2$ and keeps $a_*\beta\leq1/2$. The gauge law is \eqref{eq:regular-gauge-constrained-hessian} restricted to $I$ (so $\mathsf A_H$ is conjugated by $R_{u,*}$) applied to each block; the weights and domains do not involve the background.
\end{proof}

Put $E_0=E(Z_0)=\{b\in I\, | \,b\cap Z_0\ne\varnothing\}$, let $\iota_{E_0}$ be extension by zero and $\Pi_{I\setminus E_0}$ the complementary orthogonal projection, and set
\begin{equation}\label{eq:fixed-kernel-C-Gamma}
    C_{E_0}=(\iota_{E_0}^*H\iota_{E_0})^{-1},\qquad\Gamma_{E_0}(\sigma)=\iota_{E_0}^*H\Pi_{I\setminus E_0}S_{Z_0}(\sigma)\;.
\end{equation}
Pairings $\langle\cdot,\cdot\rangle$ involving complex vectors are the complex-bilinear extensions of the real pairing throughout.

\begin{lemma}[Gaussian conditioning]\label{lem:conditioning}
Let $\mu_0$ be the standard Gaussian measure on $\mathfrak g^I$ and $\mu_{C_{E_0}}$ the centred Gaussian measure on $\mathfrak g^{E_0}$ of covariance $C_{E_0}$. For bounded Borel $F$ on $\mathfrak g^{E_0}$,
\begin{equation}\label{eq:c-conditioning}
    \int F(a|_{E_0})\,\dd\mu_{\mathsf G}(a)=\int\left[\int F(B)\,e^{-\frac12\langle\Gamma_{E_0}(1)X,C_{E_0}\Gamma_{E_0}(1)X\rangle}e^{-\langle B,\Gamma_{E_0}(1)X\rangle}\,\dd\mu_{C_{E_0}}(B)\right]\dd\mu_0(X)\;.
\end{equation}
\end{lemma}

\begin{proof}
Write $a=\mathsf G^{1/2}X$, so that $a$ has law $\mu_{\mathsf G}$ when $X$ has law $\mu_0$. Given $a|_{I\setminus E_0}$, the conditional law of $a|_{E_0}$ under $\mu_{\mathsf G}$ is Gaussian with covariance $C_{E_0}$ and mean $-C_{E_0}\iota_{E_0}^*H\Pi_{I\setminus E_0}a=-C_{E_0}\Gamma_{E_0}(1)X$ (complete the square in $\frac12\langle a,Ha\rangle$), and its density with respect to $\mu_{C_{E_0}}$ is the displayed factor. This is RG2 (2.5)--(2.6).
\end{proof}

The weakened density is
\begin{equation}\label{eq:fixed-kernel-density}
    e^{-\frac{1}{2}\langle\Gamma_{E_0}(\sigma)X,C_{E_0}\Gamma_{E_0}(\sigma)X\rangle}e^{-\langle B,\Gamma_{E_0}(\sigma)X\rangle}\,\dd\mu_{C_{E_0}}(B)\,\dd\mu_0(X)\;.
\end{equation}
$C_{E_0}$ is real, positive and independent of $\sigma$; its determinant is one fixed positive number on the whole polydisc.

\begin{lemma}[component blocks]\label{lem:component-blocks}
Let $Z\supseteq Z_0'$ be a union of $LM$-cubes and set $\sigma_\Delta=0$ for $\Delta\not\subseteq Z$. For each wall-component $D$ of $Z$ put $J_D=\{b\in I\, | \,b\subset\operatorname{int}D\}$ and $E_D=E_0\cap J_D$. Then $E_0\subseteq\bigsqcup_DJ_D$, $E_D=E(Z_0\cap D)$, initial vertices in different $E_D$ have distance at least $2M-2>R$, and
\begin{equation}\label{eq:fixed-kernel-exact-blocks}
    C_{E_0}=\bigoplus_D(\iota_{E_D}^*H\iota_{E_D})^{-1},\qquad\Gamma_{E_0}(\sigma)=\bigoplus_D\Gamma_{E_D}(\sigma|_D)\;,
\end{equation}
where $\Gamma_{E_D}(\sigma|_D)$ is constructed from the seed $Z_0\cap D$ alone, with final domain $D$ (so $\sigma_\Delta=0$ for every cube $\Delta$ of $\pi_1$ not in $D$) and the same $I$, $H$ and background. Columns of $\Gamma_{E_0}(\sigma)$ outside $\bigsqcup_DJ_D$ vanish, and $\Gamma_{E_0}(\sigma)$ does not depend on $\sigma_\Delta$ for cubes $\Delta$ in a component containing no seed cube.
\end{lemma}

\begin{proof}
A point is interior to a union of closed grid cubes exactly when all its incident cubes belong to it, and the incident family is wall connected (any two cubes containing a point are joined by a chain of cubes containing that point, consecutive ones sharing a wall); hence $\operatorname{int}Z=\bigsqcup_D\operatorname{int}D$, a disjoint union of open sets. For an $M$-cube $Q$ of $Z_0$ the open neighbourhood $\operatorname{int}Q^\sim$ is connected and lies in $\operatorname{int}Z_0'\subseteq\operatorname{int}Z$, so it lies in one component interior; neighbourhoods in different components are disjoint, so their seed cubes have distance at least $2M$, and a bond meeting a seed cube lies in the same component interior. This gives $E_0\subseteq\bigsqcup J_D$, $E_D=E(Z_0\cap D)$ and the separation, which by finite range makes $\iota_{E_0}^*H\iota_{E_0}$ block diagonal. Consider an entry $(\Gamma_{E_0}(\sigma))_{eb}=\sum_{b'}H_{eb'}(\Pi_{I\setminus E_0})_{b'b'}S_{Z_0}(\sigma)_{b'b}$ with $e\in E_D$. A nonzero term has $d(e_-,b'_-)\leq R$; $e$ meets a seed cube $Q$, so $e_-$ is within $1$ of $Q$ and the open ball of radius $R+2$ about $e_-$ lies in the open $M$-neighbourhood $\operatorname{int}Q^\sim$ of $Q$, because $R+3\leq M$ by \eqref{eq:fixed-kernel-collar-size}, hence in $\operatorname{int}D$; so $b'\subset\operatorname{int}D$. A walk from $b'$ to $b$ with $w_\omega\ne0$ has every cube of $\pi_1$ meeting $D(\omega)$ inside $Z$, so $D(\omega)\subset\operatorname{int}Z$; $D(\omega)$ is connected and contains $b'$, so $D(\omega)\subset\operatorname{int}D$ and $b\in J_D$. Its coefficients, domains and weakening variables are independent of the seed; the cubes with $\sigma_\Delta=1$ inside $D$ are those of $Z_0'\cap D=(Z_0\cap D)'$ ($Z_0'$ is the union over the seed cubes $Q$ of the $LM$-cubes meeting $\operatorname{int}Q^\sim$; such a cube meets $\operatorname{int}Z$ at a point whose incident $LM$-cubes all lie in $Z$ and form a wall-connected family, so it lies in the component containing $Q$); and $\Pi_{I\setminus E_0}$ agrees with $\Pi_{I\setminus E_D}$ on $J_D$. This gives the block decomposition and the vanishing of the columns outside $\bigsqcup J_D$. Finally, a walk contributing to a row $e\in E_D$ has $D(\omega)\subset\operatorname{int}D$, so its weight involves only $\sigma_\Delta$ with $\Delta\subseteq D$; a component without seed cube has no row at all.
\end{proof}

The estimates on this kernel are collected in one statement.

\begin{lemma}[Gaussian payment]\label{lem:gaussian-payment}
Let $N_M=\lfloor M/(2R_*)\rfloor$, $\delta_M=2C_{\mathrm w}4^{-N_M}$ and $A_*=h_+(\gamma^{-1/2}+1)$; by \eqref{eq:c-standing-conditions}, $\delta_M\leq1$ and $h_+^2\delta_M^2/\gamma\leq1/8$. Write $\pi_i(X)$ for the set of $\pi_i$-cubes of a union $X$ of such cubes. On every branch, for every seed $Z_0$, every union $Z\supseteq Z_0'$ of $LM$-cubes with $\sigma_\Delta=0$ for $\Delta\not\subseteq Z$, and every $\sigma$ in the closed polydisc $\{|\sigma_\Delta|\leq e^{\kappa_1}\ \forall\Delta\in\pi_1(Z)\setminus\pi_1(Z_0')\}$:
\begin{enumerate}
\item $\|\Gamma_{E_0}(\sigma)-\Gamma_{E_0}(1)\|\leq h_+\delta_M$ and $\|C_{E_0}^{1/2}(\Gamma_{E_0}(\sigma)-\Gamma_{E_0}(1))\|\leq h_+\gamma^{-1/2}\delta_M$, in the operator norms for the Hilbert norms on $\mathfrak g^I_{\mathbb C}$ and $\mathfrak g^{E_0}_{\mathbb C}$.
\item For $0\leq\alpha_G\leq\min(\gamma/2,\gamma^2/(16A_*^2))$ and $J_Z=\bigsqcup_DJ_D$, the integral of the absolute value of \eqref{eq:fixed-kernel-density} against $e^{\alpha_G\|B\|^2/2}$ is at most $\exp(C_{\mathrm d}(\alpha_G+\delta_M^2)|J_Z|)$, $C_{\mathrm d}=C_{\mathrm d}(\gamma,h_+,d_G)$; with $4C_{\mathrm d}\delta_M^2(LM+1)^4\leq1/2$ (\eqref{eq:c-standing-conditions}) and $\alpha_G\leq(8C_{\mathrm d}(LM+1)^4)^{-1}$ this is at most $e^{N_1(Z)}$, and the integral of the absolute value factors over the components $D$ of $Z$ (the signed density factors by Lemma~\ref{lem:component-blocks}).
\item For $0<\gamma_2$ and the constant $c_{\mathrm n}>0$ of the group with $h(X)^2\geq c_{\mathrm n}|X|^2$ on $\mathfrak g$, $\prod_{b\in\mathsf P}\mathbf 1_{\{|B_b|\geq r\}}\leq e^{-a_{\mathrm{bad}}|\mathsf P|}\exp(\frac{\gamma_2}{2}\sum_{b\in\mathsf P}h(B_b)^2)$ with $a_{\mathrm{bad}}=c_{\mathrm n}\gamma_2r^2/2$, for every $\mathsf P\subseteq E_0$ (a pointwise inequality, used in the appendix with $\gamma_2\leq\alpha_G$).
\end{enumerate}
\end{lemma}

The proof is Appendix~\ref{app:background-estimates}, Section~\ref{app:fixed-kernel-payment}.

\subsection{The selected integrals and the polymer representation}
\label{sec:selected-integrals}

Connectivity of cubes is wall connectivity (RG1 before (0.24)); two domains are compatible when they share neither a cube nor a wall (RG2 (2.11)); $N_i(Y)$ is the number of $\pi_i$-cubes of $Y$, $d_i(Y)$ its tree length (RG1 p.257, written out in \eqref{eq:c2-tree-length}) and $\ell_i(Y)=d_i(Y)+5$. The localisation family $\mathbf D_0$ consists of the potential hosts \eqref{eq:part1-potential-host} and the probe hosts \eqref{eq:c-probe-host} below; all are connected unions of $M$-cubes. A potential family indexed by $\mathbf D_0$ is a collection $\{\mathbf W(Y,\cdot)\}_{Y\in\mathbf D_0}$ of Borel functions $\mathfrak g^I\to\mathbb C$, each depending on $a$ only through $a|_{E(Y)}$, $E(Y)=\{b\in I\, | \,b\cap Y\ne\varnothing\}$, and bounded on the local box $\{|a_b|<r\ \forall b\in E(Y)\}$; only the values on the local boxes enter below, because every selected host $Y$ comes with the factors $\chi_b$, $b\in E(Y)\subseteq E(Y_0)$. The potential $\mathbf V_\xi$ of \eqref{eq:c-potential-definition} is such a family; $\mathbf V_\xi(Y,\cdot)=0$ on hosts carrying no near primitive.

\begin{definition}[selections and selected integrals]\label{def:selection}
A selection is a triple $\mathfrak s=(\mathcal D,\mathsf P,Z)$: a finite $\mathcal D\subseteq\mathbf D_0$, a finite $\mathsf P\subseteq I\setminus E(Y_0)$ where $Y_0=\bigcup_{Y\in\mathcal D}Y$, and a final domain $Z$. Its seed is $Z_0=Y_0\cup\bigcup\{Q\in\pi_0\, | \,Q\cap b\ne\varnothing\text{ for some }b\in\mathsf P\}$, with $E_0=E(Z_0)$ and $Z_0'$ as in Section~\ref{sec:fixed-first-kernel}, and a final domain is a union $Z\supseteq Z_0'$ of $LM$-cubes. The selected integral of a bounded Borel $F$ on $\mathfrak g^{E_0}$ is
\begin{equation}\label{eq:c-selected-integral}
    \begin{aligned}
    \mathcal A^{\mathrm f}_{\mathfrak s}[F]
        &=\prod_{\Delta\in\pi_1(Z)\setminus\pi_1(Z_0')}\int_0^1\dd s_\Delta\oint_{|\sigma_\Delta|=e^{\kappa_1}}\frac{\dd\sigma_\Delta}{2\pi i(\sigma_\Delta-s_\Delta)^2}
          \prod_{Y\in\mathcal D}\int_0^1\dd t_Y\int\dd\mu_0(X)\int\dd\mu_{C_{E_0}}(B)\\
        &\quad\cdot e^{-\frac12\langle\Gamma_{E_0}(\sigma)X,C_{E_0}\Gamma_{E_0}(\sigma)X\rangle}e^{-\langle B,\Gamma_{E_0}(\sigma)X\rangle}
          (-1)^{|\mathsf P|}\prod_{b\in E(Y_0)}\chi_b(B)\prod_{b\in\mathsf P}(1-\chi_b(B))
          \prod_{Y\in\mathcal D}\mathbf W(Y,B)e^{t_Y\mathbf W(Y,B)}\,F(B)\;,
    \end{aligned}
\end{equation}
with $\chi_b(B)=\mathbf 1_{\{|B_b|<r\}}$ in CRE's bond norm (the indicator of \eqref{eq:native-source-young-cutoff}) and $\sigma_\Delta=0$ for $\Delta\not\subseteq Z$. The vacuum activity of a connected final domain $Z$ is
\begin{equation}\label{eq:c-vacuum-activity}
    k^{\mathrm f}_\xi(Z)=\sum_{(\mathcal D,\mathsf P)\, | \,Z_0\ne\varnothing,\ Z_0'\subseteq Z}\mathcal A^{\mathrm f}_{(\mathcal D,\mathsf P,Z)}[1]\;,
\end{equation}
and $\Xi^{\mathrm f}_\xi=\sum_{\mathcal P\text{ compatible}}\prod_{Z\in\mathcal P}k^{\mathrm f}_\xi(Z)$ (empty family included), $(\Xi^{\mathrm f}_\xi)^{[Y]}$ the same sum over families compatible with $Y$.
\end{definition}

The selected integral is finite: the integrand is holomorphic in $\sigma$ on the polydisc of Lemma~\ref{lem:walk-series} for fixed $(X,B,t)$, its absolute value is integrable in $(X,B)$ uniformly on the closed polydisc by Lemma~\ref{lem:gaussian-payment}(ii) together with the boundedness of $F$ and of each $\mathbf W(Y,\cdot)$ on its local box (enforced by the factors $\chi_b$, $b\in E(Y_0)$), and the contours have positive distance from $[0,1]$. Every selected datum depends only on $B$, since $E(Y)\subseteq E_0$ for $Y\in\mathcal D$ and $\mathsf P\subseteq E_0$. A fine bond meets at most two $M$-cubes (no site lies on a face), so at most $2|\mathsf P|$ cubes of the seed are not cubes of $Y_0$. The seed of a union of selections (disjoint host families and bad-bond sets) is the union of the seeds, and $\widetilde Z_0$, $Z_0'$ are unions over the seed cubes (Section~\ref{sec:fixed-first-kernel}), so all three commute with unions.

\begin{lemma}[polymer representation]\label{lem:polymer}
Let $\mathbf W$ be a potential family indexed by $\mathbf D_0$ and $\mathfrak n_\xi=(2\pi)^{d_G|I|/2}\det(H)^{-1/2}$. Then
\begin{equation}\label{eq:c-polymer}
    \int\Theta_\xi(a)\exp\Bigl(-\frac12\langle a,Ha\rangle+\sum_{Y\in\mathbf D_0}\mathbf W(Y,a)\Bigr)\dd a=\mathfrak n_\xi\,\Xi^{\mathrm f}_\xi[\mathbf W]\;,
\end{equation}
with the activities of Definition~\ref{def:selection} formed from $\mathbf W$, and the sum $\Xi^{\mathrm f}_\xi[\mathbf W]$ is finite (finitely many selections at fixed volume). In particular, for $\mathbf W=\mathbf V_\xi$,
\begin{equation}\label{eq:fixed-kernel-source-numerator}
    I_{\xi;0}=\mathfrak n_\xi\,\Xi^{\mathrm f}_\xi\;,\qquad\Xi^{\mathrm f}_\xi=\frac{I_{\xi;0}}{\mathfrak n_\xi}>0\;.
\end{equation}
\end{lemma}

\begin{proof}
The left side is $\mathfrak n_\xi\int\Theta_\xi e^{\sum_Y\mathbf W(Y,\cdot)}\dd\mu_{\mathsf G}$. Expand each $e^{\mathbf W(Y)}=1+\int_0^1\mathbf W(Y)e^{t_Y\mathbf W(Y)}\dd t_Y$; this gives the sum over $\mathcal D$. Write $\Theta_\xi=\prod_{E(Y_0)}\chi_b\prod_{I\setminus E(Y_0)}(1-(1-\chi_b))$ and expand the second product; this gives the sum over $\mathsf P$ with the sign $(-1)^{|\mathsf P|}$. The remaining integrand depends on $a|_{E_0}$ only (each $\mathbf W(Y,\cdot)$, $Y\in\mathcal D$, through $a|_{E(Y)}$ with $E(Y)\subseteq E_0$, and it is evaluated on the local box of $Y$ because of the factors $\chi_b$, $b\in E(Y_0)$), so Lemma~\ref{lem:conditioning} applies and produces \eqref{eq:fixed-kernel-density} at $\sigma=1$. For fixed $(X,B,t)$ the density \eqref{eq:fixed-kernel-density} is holomorphic in $\sigma$ on the polydisc (Lemma~\ref{lem:walk-series}); for each $\Delta\in\pi_1$ not in $Z_0'$ write its value at $\sigma_\Delta=1$ as the value at $\sigma_\Delta=0$ plus $\int_0^1\partial_{s_\Delta}\dd s_\Delta$, expand the product over these cubes, label each term by $Z=Z_0'\cup\{\text{differentiated cubes}\}$, and represent each derivative by its Cauchy integral along $|\sigma_\Delta|=e^{\kappa_1}$; these are pointwise identities in $(X,B,t)$, and the resulting parameter integrals may be taken outside the Gaussian integrals by Fubini, since the absolute value of the density is integrable uniformly on the polydisc (Lemma~\ref{lem:gaussian-payment}(ii)) and the other factors are bounded. This is \eqref{eq:c-selected-integral}; every step is an exact identity at fixed volume. The infinite objects are the walk series $S_{Z_0}(\sigma)$, which converges absolutely by Lemma~\ref{lem:walk-series}, and the Gaussian integrals over $(X,B)$ at complex $\sigma$, whose absolute convergence is Lemma~\ref{lem:gaussian-payment}(ii) (the walk-series bound alone does not control the real part of the quadratic exponent). Every cube of $Z_0'$ is wall-connected inside $Z_0'$ to a cube containing a seed cube: a cube $\Delta$ of $Z_0'$ meets $\operatorname{int}Q^\sim$ for a seed cube $Q$ at a point $x$; since $\operatorname{int}Q^\sim$ is open, $x$ may be taken in the interior of $\Delta$, so $x$ lies in an $M$-cube $Q'\subseteq Q^\sim\cap\Delta$, and $Q'$ is joined to $Q$ by a chain of face-adjacent $M$-cubes of $Q^\sim$, whose $LM$-cubes start with $\Delta$, are consecutively equal or wall adjacent, and all belong to $Z_0'$ (each contains an $M$-cube of $Q^\sim$, whose interior lies in $\operatorname{int}Q^\sim$). Hence a wall-component of $Z$ either contains a seed cube or consists of differentiated cubes only; in the second case $\Gamma_{E_0}(\sigma)$ does not depend on its $\sigma$-variables (Lemma~\ref{lem:component-blocks}), the Cauchy integral of a function constant in $\sigma_\Delta$ vanishes, and the term is zero. So only final domains all of whose components contain a seed cube contribute. For these, Lemma~\ref{lem:component-blocks} factors the density and the Gaussian measures over the components $D$ of $Z$ ($\mu_{C_{E_0}}=\bigotimes_D\mu_{C_{E_D}}$ and the density is a product by the block decomposition; the $X$-integral factors because each block of $\Gamma_{E_0}$ reads $X$ on $J_D$ only, and the remaining Gaussian factor of $\mu_0$ integrates to one), and the parameter integrals are indexed by cubes and hosts, each inside one component. The cutoff, potential and sign factors factor because each selected host is a connected union of $M$-cubes inside $Z_0$, hence inside one component, and each bad bond lies in the component containing its (at most two, face-adjacent) incident cubes. Given $Z$ with components $D_1,\ldots,D_m$, the map $(\mathcal D,\mathsf P)\mapsto((\mathcal D_j,\mathsf P_j))_j$, with $\mathcal D_j$ the hosts inside $D_j$ and $\mathsf P_j$ the bad bonds inside $D_j$, is a bijection from the selections with final domain $Z$ (all $Z_0'\subseteq Z$, every component seeded) onto the tuples of selections $(\mathcal D_j,\mathsf P_j,D_j)$ with nonempty seeds and final domains $D_j$: the inverse is the union, because the seed rule, the enlargement and the coarse cover commute with unions and $(Z_0\cap D_j)'\subseteq D_j$ (it is wall-connected to its seed inside $Z_0'\subseteq Z$). Summing over $Z$ gives the compatible-family sum, in which the empty selection contributes $\int\dd\mu_{\mathsf G}=1$. Equation \eqref{eq:fixed-kernel-source-numerator} is the case $\mathbf W=\mathbf V_\xi$: $K_{\xi;0}=\Theta_\xi\exp(-\frac12\langle a,Ha\rangle+\sum_Y\mathbf V_\xi(Y,a))$ by \eqref{eq:native-source-boundary-interpolation} and \eqref{eq:c-potential-definition} (on the open box $\sum_Y\mathbf V_\xi(Y,a)=P_{\xi,0}(a)$), and positivity holds because $K_{\xi;0}>0$ on the open box.
\end{proof}

\begin{lemma}[activity bounds]\label{lem:activity-bounds}
With the constants $\lambda_*,\varepsilon_*>0$ and $\theta\geq0$ of Appendix~\ref{app:background-estimates}, Section~\ref{app:mayer-constants} (fixed there before the coupling together with its auxiliary constants $\kappa,\delta,\alpha_6,\omega_*$, independently of $M_1$ and $M$), and $g_0$ below its threshold, for $\mathbf W=\mathbf V_\xi$ on every branch and every connected final domain $Z$:
\begin{equation}\label{eq:fixed-kernel-raw-activity-payment}
    |k^{\mathrm f}_\xi(Z)|\leq\varepsilon_*e^{-\lambda_*\ell_1(Z)},\qquad
    \sum_{Z'\not\sim Z}|k^{\mathrm f}_\xi(Z')|e^{(1+6\theta)N_1(Z')}\leq\frac{1}{8}N_1(Z)\;,
\end{equation}
where $Z'$ ranges over the connected final domains and $Z'\not\sim Z$ means incompatible (a domain is incompatible with itself). The same holds for the reference data of Section~\ref{sec:joint-extraction} (Lemma~\ref{lem:c2-reference}).
\end{lemma}

\begin{lemma}[vacuum clusters]\label{lem:vacuum-clusters}
Under Lemma~\ref{lem:activity-bounds}, for an ordered tuple $\mathfrak c=(D_1,\ldots,D_m)$, $m\geq1$, of connected final domains with Ursell coefficient $\varphi^T(\mathfrak c)=\sum_{\mathcal G}(-1)^{|\mathcal G|}$, the sum over the connected spanning subgraphs $\mathcal G$ of the incompatibility graph on $\{1,\ldots,m\}$ ($j\ne j'$ joined when $D_j\not\sim D_{j'}$; a domain is incompatible with itself, so repeated domains are joined; $\varphi^T=1$ for $m=1$), put
\begin{equation}\label{eq:part1-vacuum-cluster}
    \Phi^{\mathrm v}_\xi(\mathfrak c)=\frac{\varphi^T(\mathfrak c)}{m!}\prod_{j=1}^mk^{\mathrm f}_\xi(D_j)\;.
\end{equation}
For every finite set $\mathcal A$ of connected final domains, $\sum_{\mathfrak c\text{ hits }\mathcal A}|\Phi^{\mathrm v}_\xi(\mathfrak c)|e^{2\theta\sum_jN_1(D_j)}\leq\frac18\sum_{D\in\mathcal A}N_1(D)$, where $\mathfrak c$ hits $\mathcal A$ when some $D_j$ is incompatible with a member of $\mathcal A$, and
\begin{equation}\label{eq:part1-joint-vacuum-exponential}
    \frac{(\Xi^{\mathrm f}_\xi)^{[\cup\mathcal A]}}{\Xi^{\mathrm f}_\xi}=\exp\Bigl(-\sum_{\mathfrak c\text{ hits }\mathcal A}\Phi^{\mathrm v}_\xi(\mathfrak c)\Bigr)\;.
\end{equation}
\end{lemma}

The proof, including the identity \eqref{eq:part1-joint-vacuum-exponential}, is Appendix~\ref{app:background-estimates}, Section~\ref{app:kp}: the Kotecký--Preiss criterion with the pinned-tree bound.

\subsection{Probe kernels and the centred expectation}
\label{sec:probe-kernels}

For each $b\in I$ let $\mathcal R_b=\{r\text{ a bond of }\Omega_1\, | \,b\in\mathcal F_r\}$ be its forward influence set (the bonds whose reconstructed link depends on $a(b)$, Lemma~\ref{lem:row-locality}); it lies within $\ell^\infty$ distance $2L+1$ of $b$. Lift $\{b\}\cup\mathcal R_b$ to $\mathbb Z^4$ about $b_-$ (its diameter is below every period), and let
\begin{equation}\label{eq:c-probe-host}
    Y_b=\pi_T\bigl(\text{the smallest rectangle of $M$-cubes containing the one-fine-layer neighbourhood of the lift, with one $M$-layer adjoined}\bigr)\;;
\end{equation}
the lift together with its one-fine-layer neighbourhood has $\ell^\infty$ extent at most $2(2L+1)+3=4L+5<M$ in every direction ($M\geq8LM_2\geq8L^2$), so it meets at most two $M$-cubes per direction and the rectangle has side at most $4$ cubes, of length $4M<8M$, so $\pi_T$ is injective on it. $Y_b$ is a connected member of $\mathbf D_0$, contains $b$ and $\mathcal R_b$ in its interior, and $N_0(Y_b)\leq4^4\leq s_{\mathrm p}=5^4$. For a union $R$ of $LM$-cubes its eligible labels are
\begin{equation}\label{eq:probe-eligible-carrier}
    \mathsf L_R=\{b\in I\, | \,Y_b\subseteq R\}\;,
\end{equation}
and $\overline{\mathcal D}_{\mathsf L}$ is the closed box $|a_b|\leq r$, $b\in\mathsf L$, in $\mathfrak g^{\mathsf L}$. Each $Y_b$ is a nonempty wall-connected union of $M$-cubes, so $Y_b\subseteq R$ places it inside exactly one wall-component of $R$; hence $\mathsf L_R$ is the disjoint union of the $\mathsf L_D$ over the components $D$ of $R$. For $\lambda\in\mathfrak g^I_{\mathbb C}$, with the complex bilinear extension of the real pairing, the probe family is
\begin{equation}\label{eq:probe-combined-potential}
    L_Y(\lambda,a)=\sum_{b\, | \,Y_b=Y}\langle\lambda_b,a_b\rangle,\qquad\widetilde{\mathbf V}_{\xi,\lambda}(Y,a)=\mathbf V_\xi(Y,a)+L_Y(\lambda,a)\;,
\end{equation}
a potential family indexed by $\mathbf D_0$ for each $\lambda$ (both terms depend on $a|_{E(Y)}$ only, since $Y_b=Y$ gives $b\in E(Y)$, and both are bounded on the local box of $Y$), entire in $\lambda$. Its partition function is
\begin{equation}\label{eq:probe-partition}
    \mathscr Z^{\mathrm c}_\xi(\lambda)=\int e^{\sum_{b\in I}\langle\lambda_b,a_b\rangle}K_{\xi;0}(a)\,\dd a=\mathfrak n_\xi\sum_{\mathcal P\text{ compatible}}\prod_{D\in\mathcal P}k^{\mathrm f}_{\xi,D}(\lambda)\;,
\end{equation}
by Lemma~\ref{lem:polymer} applied to $\widetilde{\mathbf V}_{\xi,\lambda}$, where $k^{\mathrm f}_{\xi,D}(\lambda)$ denotes the activity formed from that family; it is entire in $\lambda$ (a finite sum of integrals of entire functions of $\lambda$ with integrable majorants on compact sets of $\lambda$) and $k^{\mathrm f}_{\xi,D}(0)=k^{\mathrm f}_\xi(D)$.

\begin{lemma}[probe measures]\label{lem:probe-measures}
On every branch, for every connected final domain $D$ there is a finite complex Borel measure $\nu^{\mathrm f}_{\xi,D}$ on $\overline{\mathcal D}_{\mathsf L_D}$ with
\begin{equation}\label{eq:probe-Laplace-identity}
    k^{\mathrm f}_{\xi,D}(\lambda)=\int e^{\sum_{b\in\mathsf L_D}\langle\lambda_b,a_b\rangle}\,\dd\nu^{\mathrm f}_{\xi,D}(a)\quad(\lambda\in\mathfrak g^{I}_{\mathbb C}),\qquad
    \nu^{\mathrm f}_{\xi,D}(\overline{\mathcal D}_{\mathsf L_D})=k^{\mathrm f}_\xi(D)\;;
\end{equation}
in particular $k^{\mathrm f}_{\xi,D}(\lambda)$ depends on $\lambda$ only through $\lambda|_{\mathsf L_D}$. Explicitly, splitting each probe label from its host's ordinary label, a selection becomes $\ell=(\mathcal D,\mathsf J,\mathsf P,Z)$ with $\mathsf J\subseteq\mathsf L_Z$ the selected probes, $Y_0=\bigcup_{Y\in\mathcal D}Y\cup\bigcup_{b\in\mathsf J}Y_b$, and seed, $E_0$ and $Z_0'$ formed from this $Y_0$ and $\mathsf P$; the Mayer variable of a probe integrates exactly,
\begin{equation}\label{eq:probe-selected-factor}
    \int_0^1\langle\lambda_b,B_b\rangle e^{t_b\langle\lambda_b,B_b\rangle}\,\dd t_b=e^{\langle\lambda_b,B_b\rangle}-1\;,
\end{equation}
and with $\Lambda^{\mathrm f}_{\xi;\ell}$ the complex measure on $\mathfrak g^{E_0}$ defined by the right side of \eqref{eq:c-selected-integral} for $\mathbf W=\mathbf V_\xi$, with $Y_0$, $Z_0$, $E_0$, $Z_0'$ the enlarged data of $\ell$ throughout (the factors $\chi_b$ over $E(Y_0)$, the conditional law and the weakening over this seed), with the Mayer factors $\int_0^1\dd t_Y\,\mathbf W(Y,B)e^{t_Y\mathbf W(Y,B)}$ for $Y\in\mathcal D$ only and no probe factor,
\begin{equation}\label{eq:probe-selection-measure}
    \nu^{\mathrm f}_{\xi,D}=\sum_{\ell\, | \,Z=D,\ Z_0\ne\varnothing}\int\Bigl[\bigotimes_{b\in\mathsf J}(\delta_{B_b}-\delta_0)\otimes\bigotimes_{b\in\mathsf L_D\setminus\mathsf J}\delta_0\Bigr]\,\dd\Lambda^{\mathrm f}_{\xi;\ell}(B)\;.
\end{equation}
\end{lemma}

\begin{proof}
At a host $Y$ carrying the ordinary potential $\mathbf V_\xi(Y)$ and the probe terms $F_b=\langle\lambda_b,\cdot\rangle$, $b$ with $Y_b=Y$, write $e^{\widetilde{\mathbf V}_{\xi,\lambda}(Y)}=e^{\mathbf V_\xi(Y)}\prod_be^{F_b}=\prod_{\ell'}(1+(e^{F_{\ell'}}-1))$ over the labels $\ell'$ of $Y$ (the ordinary label with $F_{\ell'}=\mathbf V_\xi(Y)$ and one label per probe); the sum over nonempty label subsets $A$ of $\prod_{\ell'\in A}(e^{F_{\ell'}}-1)$ equals $e^{\widetilde{\mathbf V}_{\xi,\lambda}(Y)}-1$. Every term with $A\ne\varnothing$ has the same active host $Y$, so the split-label expansion has the same $Y_0$, $\mathsf P$, seed, cover and kernel as the one-potential-per-host expansion of Lemma~\ref{lem:polymer}, and the two activities agree term by term after the Mayer variables $t_b$ of the probes are integrated by \eqref{eq:probe-selected-factor}. The Laplace transform of $\delta_{B_b}-\delta_0$ in the variable $\lambda_b$ is $e^{\langle\lambda_b,B_b\rangle}-1$, and an unselected label contributes the factor one, the transform of $\delta_0$; this proves \eqref{eq:probe-Laplace-identity} for the measure \eqref{eq:probe-selection-measure}. Each $\Lambda^{\mathrm f}_{\xi;\ell}$ is a finite complex measure: it is the image under $(X,B,s,\sigma,t)\mapsto B$ of a complex density whose absolute value is integrable by Lemma~\ref{lem:gaussian-payment}(ii) and the boundedness of each $\mathbf V_\xi(Y,\cdot)$ on its local box, the Cauchy kernels and the cutoff factors; the Dirac tensor has total variation at most $2^{|\mathsf J|}$, and there are finitely many $\ell$ at fixed volume. The support lies in the closed box because $\chi_b(B)$, $b\in E(Y_0)\supseteq\mathsf J$, forces $|B_b|<r$ on the selected probes. At $\lambda=0$ every term with $\mathsf J\ne\varnothing$ has zero mass, so $\nu^{\mathrm f}_{\xi,D}(\overline{\mathcal D}_{\mathsf L_D})=k^{\mathrm f}_{\xi,D}(0)=k^{\mathrm f}_\xi(D)$.
\end{proof}

For a compatible family $\mathcal P$ let $\mathsf e_{\mathcal P}$ embed $\prod_{D\in\mathcal P}\mathfrak g^{\mathsf L_D}$ into $\mathfrak g^I$ by zero on the remaining labels (the $\mathsf L_D$ are pairwise disjoint).

\begin{lemma}[measure identity]\label{lem:measure-identity}
On every branch,
\begin{equation}\label{eq:fixed-probe-measure}
    \frac{K_{\xi;0}(a)\,\dd a}{\mathfrak n_\xi}=\sum_{\mathcal P\text{ compatible}}(\mathsf e_{\mathcal P})_*\bigotimes_{D\in\mathcal P}\nu^{\mathrm f}_{\xi,D}\;.
\end{equation}
\end{lemma}

\begin{proof}
Both sides are finite complex measures with compact support (finitely many families at fixed volume), and by \eqref{eq:probe-partition} and \eqref{eq:probe-Laplace-identity} they have the same Laplace transform on $\mathfrak g^I_{\mathbb C}$; restricted to imaginary $\lambda$ this is the Fourier transform, which determines a finite measure.
\end{proof}

Let $\delta_{0,\mathsf L_D}$ be the Dirac mass at $0\in\mathfrak g^{\mathsf L_D}$ and
\begin{equation}\label{eq:fixed-probe-zero-mass}
    \widehat\nu^{\mathrm f}_{\xi,D}=\nu^{\mathrm f}_{\xi,D}-k^{\mathrm f}_\xi(D)\delta_{0,\mathsf L_D}\;,
\end{equation}
a measure of zero mass (the subtracted term is the sum of the selections with $\mathsf J=\varnothing$). For a nonempty compatible family $\mathcal A$ and bounded Borel $f$ on $\mathfrak g^I$ put
\begin{equation}\label{eq:fixed-probe-active-pairing}
    \mathfrak Q_\xi(\mathcal A;f)=\int f(\mathsf e_{\mathcal A}a)\,\dd\Bigl(\bigotimes_{D\in\mathcal A}\widehat\nu^{\mathrm f}_{\xi,D}\Bigr)(a),\qquad
    q_\xi(\mathcal A)=\frac{(\Xi^{\mathrm f}_\xi)^{[\cup\mathcal A]}}{\Xi^{\mathrm f}_\xi}\;.
\end{equation}

\begin{proposition}[centred expectation]\label{prop:centred-expectation}
On every branch, for bounded Borel $f$ on $\mathfrak g^I$,
\begin{equation}\label{eq:fixed-probe-centred-source}
    \int f\,\dd\nu_{\xi;0}=\frac{\int f(a)K_{\xi;0}(a)\,\dd a}{I_{\xi;0}}=f(0)+\sum_{\varnothing\ne\mathcal A\text{ compatible}}q_\xi(\mathcal A)\mathfrak Q_\xi(\mathcal A;f)\;,
\end{equation}
and, with Lemmas~\ref{lem:collar} and~\ref{lem:boundary-response},
\begin{equation}\label{eq:c-physical-numerator}
    \mathcal J^{\mathrm a}_\xi[f]=e^{c_\xi}I_{\xi;1}\Bigl[f(0)+\sum_{\varnothing\ne\mathcal A\text{ compatible}}q_\xi(\mathcal A)\mathfrak Q_\xi(\mathcal A;f)+\mathcal B_\xi[f]\Bigr]\;.
\end{equation}
\end{proposition}

\begin{proof}
Integrate $f$ against \eqref{eq:fixed-probe-measure} and expand each $\nu^{\mathrm f}_{\xi,D}$ as $\widehat\nu^{\mathrm f}_{\xi,D}+k^{\mathrm f}_\xi(D)\delta_{0,\mathsf L_D}$. A term is indexed by $\mathcal P$ and the subfamily $\mathcal A\subseteq\mathcal P$ carrying the zero-mass factors; $f(\mathsf e_{\mathcal P}a)$ with $a=0$ on the labels of $\mathcal P\setminus\mathcal A$ is $f(\mathsf e_{\mathcal A}a)$. Summing over $\mathcal P\supseteq\mathcal A$ gives $(\Xi^{\mathrm f}_\xi)^{[\cup\mathcal A]}\mathfrak Q_\xi(\mathcal A;f)$, and $\mathcal A=\varnothing$ gives $\Xi^{\mathrm f}_\xi f(0)$. Divide by $I_{\xi;0}=\mathfrak n_\xi\Xi^{\mathrm f}_\xi$. The second display is $\mathcal J^{\mathrm a}_\xi[f]=e^{c_\xi}I_{\xi;1}\int f\,\dd\nu_{\xi;1}$ \eqref{eq:c-collar-identity} with \eqref{eq:fixed-probe-boundary-functional}.
\end{proof}

With $f=f_\xi$ (extended by zero off the enlarged box; the measures live on the closed box, inside the enlarged box, where $f_\xi$ is the Wilson function \eqref{eq:convex-response-operands}), the first display is $F_\xi(0)$, and \eqref{eq:c-physical-numerator} is the exact decomposition of the conditional Wilson numerator on every branch: the loop at zero young field, the sum of closed insertions paired with the signed probe measures, and the exterior response.

\subsection{Joint extraction of the scalar reference}
\label{sec:joint-extraction}

Let $r_n$ be the positive representative of traversal $n\in\{1,\ldots,n_C\}$ of $C$, the traversals numbered cyclically from a fixed base point. For a contour bond $r$ that is not a bond of $\Omega_1$ put $\mathcal F_r=\varnothing$: its link is $V_0(r)$ by \eqref{eq:c-reconstructed-field}, independent of $a$. By Lemma~\ref{lem:row-locality} and this convention, the link $U_\xi(a)(r_n)$ depends on the young variable only through $a|_{\mathcal F_{r_n}\cap I}$. For a connected final domain $D$ the contact set is
\begin{equation}\label{eq:part1-contact-set}
    \mathsf T_D=\{n\, | \,\mathcal F_{r_n}\cap\mathsf L_D\ne\varnothing\}\;,
\end{equation}
and $D$ is marked when $\mathsf T_D\ne\varnothing$. If a family $\mathcal A$ contains an unmarked domain $D$, then for $a$ in the closed box $f_\xi(\mathsf e_{\mathcal A}a)=W_\chi(C;U_\xi(\mathsf e_{\mathcal A}a))$ does not depend on $a|_{\mathsf L_D}$ (every contour link depends on $a|_{\mathcal F_{r_n}\cap I}$, and $\mathcal F_{r_n}\cap\mathsf L_D=\varnothing$ for every $n$), and $\mathfrak Q_\xi(\mathcal A;f_\xi)=0$ because $\widehat\nu^{\mathrm f}_{\xi,D}$ has zero mass and the measures are supported in the closed box; so only families of marked domains contribute to \eqref{eq:fixed-probe-centred-source} for $f=f_\xi$. Expand $q_\xi(\mathcal A)$ by Lemma~\ref{lem:vacuum-clusters},
\begin{equation}\label{eq:c-cluster-expansion}
    q_\xi(\mathcal A)=\sum_{k\geq0}\frac{(-1)^k}{k!}\sum_{\mathfrak c_1,\ldots,\mathfrak c_k\text{ each hits }\mathcal A}\prod_{j=1}^k\Phi^{\mathrm v}_\xi(\mathfrak c_j)\;,
\end{equation}
the sum being absolutely convergent: $\sum_{\mathfrak c\text{ hits }\mathcal A}|\Phi^{\mathrm v}_\xi(\mathfrak c)|\leq\frac18\sum_{D\in\mathcal A}N_1(D)$ by Lemma~\ref{lem:vacuum-clusters}, so the $k$-fold sums are dominated by the exponential series. For each term form the graph on the marked domains of $\mathcal A$ and the $k$ cluster occurrences: a marked vertex is joined to each cluster that hits it, two clusters are joined when some constituent domains are incompatible. Every component has a marked root; its contacts are the union of the contact sets of its roots. The hull of a nonempty set of traversals is its shortest cyclic interval when that interval has fewer than $n_C/2$ positions, and the whole contour otherwise; a shortest interval of fewer than $n_C/2$ positions is unique (two such intervals containing the set have together fewer than $n_C$ positions, so they intersect in a cyclic interval, which contains the set and is shorter than one of them unless they coincide), and if $S\subseteq S'$ and the hull of $S'$ is short then the hull of $S$ is contained in it (the hull of $S'$ is a short interval containing $S$, so the hull of $S$ is short and, by the uniqueness just proved, is the shortest interval containing $S$; the intersection of the two short hulls is again an interval, their union having fewer than $n_C$ positions, and it contains $S$, so it is the hull of $S$). Let $\approx$ be the transitive closure of the relation ``the hulls of the contacts intersect'' on the components, and for a class $K$ let $\mathcal H(K)$ be the hull of the union of the contacts of its members. If the union of the members' hulls has fewer than $n_C/2$ positions, $\mathcal H(K)$ equals that union: the union is a cyclic interval (the members' hulls are chained by intersections, and two intersecting intervals with together fewer than $n_C$ positions have an interval union), it contains the union of the contacts, so it contains the hull $\mathcal H(K)$ by minimality; conversely each member's contacts lie in the union of the contacts, whose hull is short, so by monotonicity each member's hull lies in $\mathcal H(K)$. If the union of the members' hulls has at least $n_C/2$ positions, $\mathcal H(K)$ is the whole contour: a short $\mathcal H(K)$ would contain every member's hull by monotonicity. The groups are the classes if every $\mathcal H(K)$ is short, and the single class of all components otherwise; the intervals of distinct groups are disjoint, since intersecting intervals would relate their classes.

\begin{definition}[groups]\label{def:groups}
A group $W$ consists of its nonempty compatible marked-root set $\mathcal A(W)$, its ordered list of cluster occurrences $(\mathfrak c_1,\ldots,\mathfrak c_{k(W)})$, $k(W)\geq0$, each hitting a root of $W$, and its interval $\mathcal H(W)$, either a cyclic interval of fewer than $n_C/2$ traversals or the whole contour read from the base point; the data arise from a term of \eqref{eq:c-cluster-expansion} as above; conversely, the term of \eqref{eq:c-cluster-expansion} with root family $\mathcal A(W)$ and cluster list $(\mathfrak c_1,\ldots,\mathfrak c_{k(W)})$ has $W$ as its only group (its graph has the components of $W$, which form one class with the interval $\mathcal H(W)$). Its weight is
\begin{equation}\label{eq:part1-group-weight}
    w_\xi(W)=\frac{(-1)^{k(W)}}{k(W)!}\prod_{j=1}^{k(W)}\Phi^{\mathrm v}_\xi(\mathfrak c_j)\;.
\end{equation}
Two groups are compatible when all their constituent domains (roots and cluster members) are mutually compatible and their intervals are disjoint; a family containing a whole-contour group is a singleton. For a compatible group family $\mathfrak B$ and $\mathcal S\subseteq\mathfrak B$,
\begin{equation}\label{eq:part1-closed-family-pairing}
    \mathcal A(\mathcal S)=\bigcup_{W\in\mathcal S}\mathcal A(W),\qquad
    A_\xi(\mathcal S)=\Bigl(\prod_{W\in\mathcal S}w_\xi(W)\Bigr)\mathfrak Q_\xi(\mathcal A(\mathcal S);f_\xi),\qquad A_\xi(\varnothing)=f_\xi(0)\;.
\end{equation}
\end{definition}

\begin{lemma}[grouped source]\label{lem:grouped-source}
On every branch,
\begin{equation}\label{eq:part1-grouped-source}
    F_\xi(0)=\sum_{\mathfrak B\text{ compatible}}A_\xi(\mathfrak B)\;,
\end{equation}
an absolutely convergent sum over compatible group families (including the empty family; there are finitely many root families at fixed volume but infinitely many cluster lists).
\end{lemma}

\begin{proof}
Substitute \eqref{eq:c-cluster-expansion} into \eqref{eq:fixed-probe-centred-source} for $f=f_\xi$. Each term, indexed by a family $\mathcal A$ of marked domains and an ordered list of $k$ cluster occurrences, is cut into its groups; the groups are pairwise compatible (roots in different groups are compatible as members of $\mathcal A$, clusters in different groups are compatible with each other and with the other groups' roots because the graph has no edge between components, and the intervals are disjoint by construction). The regrouping is reversible: a compatible group family determines $\mathcal A=\mathcal A(\mathfrak B)$ and the multiset of cluster occurrences, and the term's graph has as components exactly the components of the groups' own graphs (no edges between groups, by compatibility), with the same hulls, so the classes and intervals are recovered; identical occurrences cannot split between groups because their domains are self-incompatible, so a group's occurrences are those of its term. The ordered list of the term is an interleaving of the groups' lists, and the interleavings number $k!/\prod_Wk(W)!$, which turns the factor $(-1)^k/k!$ into $\prod_Ww_\xi(W)/\prod_W\Phi$-products. The rearrangement is of an absolutely convergent sum: there are finitely many root families $\mathcal A$ at fixed volume, $|\mathfrak Q_\xi(\mathcal A;f_\xi)|\leq\|f_\xi\|_\infty\prod_{D\in\mathcal A}\|\widehat\nu^{\mathrm f}_{\xi,D}\|$ with the total variations finite, and $\sum_{k}\frac{1}{k!}\bigl(\sum_{\mathfrak c\text{ hits }\mathcal A}|\Phi^{\mathrm v}_\xi(\mathfrak c)|\bigr)^k\leq\exp(\frac18\sum_{D\in\mathcal A}N_1(D))$; hence the term sum converges absolutely, and so does the group-family sum, being a rearrangement of it.
\end{proof}

A superscript $\circ$ denotes the reference data: the same $\Omega_1$, $\Lambda_1$, $I$, $J$, cubes, hosts, constants and cutoffs, with the background $U^\circ_1=\id$ on every bond, $V^\circ_0=\id$, zero retained field $A^\circ_{\mathrm e}=0$ and $\Psi^\circ_\xi=\id$; thus $H^\circ_\xi=\mathsf H^{\Omega_1}_{\id}$, the reference phase is $q_{\id}$, the reference primitives are those of Section~\ref{sec:source-phase-primitives} at $U=\id$, $f^\circ_\xi(a)=W_\chi(C;e^{iq_{\id}(a)})$ with the links off $\Omega_1$ equal to $\id$, and $f^\circ_\xi(0)=\chi(\id)=1$ since $q_{\id}(0)=0$. These data need not be a branch of the history sum. Lemmas~\ref{lem:two-block}, \ref{lem:hessian-gauge-locality}, \ref{lem:row-locality} and~\ref{lem:primitive-sum} apply to $U=\id$ directly (Section~\ref{sec:source-phase-primitives}, opening paragraph; Lemma~\ref{lem:two-block} at $A=0$; the row solutions are those of Lemma~\ref{lem:charts} in its reference form, on the row polydiscs $\mathcal U_c$, and (B4) is not invoked), and the estimate lemmas used below (Lemmas~\ref{lem:regular-bounds}, \ref{lem:charts}, \ref{lem:gaussian-payment}, \ref{lem:activity-bounds}, \ref{lem:vacuum-clusters}) hold for the reference data by Appendix~\ref{app:background-estimates}, Lemma~\ref{lem:c2-reference}; in particular the reference near primitives are real on the open box (Lemma~\ref{lem:charts}: the analytic branches of the logarithms are real at real arguments, the row solution being real), so $K^\circ_{\xi;0}>0$ there and $I^\circ_{\xi;0}>0$. Every lemma of Sections~\ref{sec:fixed-first-kernel}--\ref{sec:joint-extraction} stated on every branch (Lemmas~\ref{lem:walk-series}, \ref{lem:conditioning}, \ref{lem:component-blocks}, \ref{lem:polymer}, \ref{lem:probe-measures}, \ref{lem:measure-identity}, \ref{lem:grouped-source} and the first display of Proposition~\ref{prop:centred-expectation}) holds verbatim for the reference data: their proofs use only the field-independent geometry and the lemmas just listed. For a group $W$ let $P^\circ_W(a)$ be the ordered product of the reference links along the consecutive traversals of $\mathcal H(W)$, in the cyclic order of the contour starting from the first traversal of the interval (for a whole-contour group, from the base point), the link of a traversal being $e^{iq_{\id}(\mathsf e_{\mathcal A(W)}a)(r)}$ when its bond $r$ is a bond of $\Omega_1$ and $\id$ otherwise, as in $f^\circ_\xi$ (oriented representatives inverted), with the young variable $\mathsf e_{\mathcal A(W)}a$, zero off the labels of $\mathcal A(W)$; then $P^\circ_W(0)=\id$. \end{equation}

\begin{lemma}[Schur identity]\label{lem:schur}
For $h\in G$,
\begin{equation}\label{eq:part1-closed-schur}
    w^\circ_\xi(W)\int\chi\bigl(P^\circ_W(a)h\bigr)\,\dd\Bigl(\bigotimes_{D\in\mathcal A(W)}\widehat\nu^{\mathrm f,\circ}_{\xi,D}\Bigr)(a)=a_W\chi(h)\;.
\end{equation}
Consequently
\begin{equation}\label{eq:part1-reference-gas}
    S^\circ_{\xi,C}=\sum_{\mathfrak B\text{ compatible}}\prod_{W\in\mathfrak B}a_W=\frac{\int f^\circ_\xi(a)K^\circ_{\xi;0}(a)\,\dd a}{I^\circ_{\xi;0}}\;.
\end{equation}
\end{lemma}

\begin{proof}
For constant $u\in G$ let $(R_ua)_b=\operatorname{Ad}_ua_b$ on $\mathfrak g^I$, an orthogonal map for the invariant inner product. At the identity background the constant gauge transformation $u$ fixes the background, so Lemma~\ref{lem:hessian-gauge-locality} gives $H^\circ_\xi=R_uH^\circ_\xi R_u^{-1}$; hence all walk coefficients, $C^\circ_{E_0}$, $\Gamma^\circ_{E_0}(\sigma)$ (the domains and weights do not involve the fields) and the Gaussian density \eqref{eq:fixed-kernel-density} are invariant under $(X,B)\mapsto(R_uX,R_uB)$, and $\mu_0$, $\mu_{C^\circ_{E_0}}$ are invariant. The cutoff factors are invariant because the bond norm is $\operatorname{Ad}$-invariant (\eqref{eq:marginal-cutoff-invariance}). The reconstruction is covariant: by the intertwinings in the proof of Lemma~\ref{lem:hessian-gauge-locality} at $U=\id$, $\mathcal N_{\id}\circ R_u=R^{(1)}_u\mathcal N_{\id}$, $h_{\id}R^{(1)}_u=R_uh_{\id}$ and $\mathsf C_{\id}R_{u,*}=R_u\mathsf C_{\id}$, so the unique small solution on the row polydiscs of Lemma~\ref{lem:charts} (reference version; the polydiscs are $R_u$-invariant because the bond norm is $\operatorname{Ad}$-invariant) satisfies $\widetilde D_{\id}(R_uA)=R^{(1)}_u\widetilde D_{\id}(A)$ and $q_{\id}(R_uB)=R_uq_{\id}(B)$, that is, every reconstructed link is conjugated by $u$; the primitives are invariant, being traces of products of links, Haar densities of conjugated phases ($\sigma$ is conjugation invariant) and determinants of conjugated operators. The probe term transforms as $L_Y(R_u\lambda,R_ua)=L_Y(\lambda,a)$. Hence $k^{\mathrm f,\circ}_{\xi,D}(R_u\lambda)=k^{\mathrm f,\circ}_{\xi,D}(\lambda)$, and by Laplace uniqueness $(R_u)_*\widehat\nu^{\mathrm f,\circ}_{\xi,D}=\widehat\nu^{\mathrm f,\circ}_{\xi,D}$. Since the links are conjugated, $P^\circ_W(R_ua)=uP^\circ_W(a)u^{-1}$. Let $\pi$ be the irreducible unitary representation with $\chi=\dim(\pi)^{-1}\operatorname{tr}\pi$ (the normalised character, $\chi(\id)=1$). The left side of \eqref{eq:part1-closed-schur}, as a function of $h$, is $\dim(\pi)^{-1}\operatorname{tr}(\Upsilon\pi(h))$ for the matrix $\Upsilon=w^\circ_\xi(W)\int\pi(P^\circ_W(a))\,\dd(\bigotimes\widehat\nu^\circ)(a)$; by the invariance of the measure and the covariance of $P^\circ_W$, $\Upsilon=\pi(u)\Upsilon\pi(u)^{-1}$ for all $u\in G$, so $\Upsilon$ is a multiple of the identity by Schur's lemma, and $h=\id$ identifies the multiple as $a_W$. For \eqref{eq:part1-reference-gas} apply Lemma~\ref{lem:grouped-source} to the reference data (its sum converges absolutely, as shown there). Let $\mathfrak B=\{W_1,\ldots,W_m\}$ be a compatible family. If $m=1$ and $W_1$ is a whole-contour group, the reference Wilson function at $\mathsf e_{\mathcal A(W_1)}a$ is $\chi(P^\circ_{W_1}(a))$. Otherwise all intervals are short and pairwise disjoint; number the groups in cyclic order starting from $W_1$, and use the cyclicity of the trace to start the ordered link product at the first traversal of $\mathcal H(W_1)$. The reference Wilson function at $\mathsf e_{\mathcal A(\mathfrak B)}a$ is then $\chi(P^\circ_{W_1}(a)\cdots P^\circ_{W_m}(a))$: a traversal outside every interval is outside every contact set, so its link is $e^{iq_{\id}(0)}=\id$ (or $\id$ by convention when its bond is not a bond of $\Omega_1$); a traversal in $\mathcal H(W_j)$ lies in no contact set of a root of another group, so its link depends only on $a|_{\mathcal A(W_j)}$, and the links of the consecutive traversals of $\mathcal H(W_j)$ multiply to $P^\circ_{W_j}(a)$ in the order of the product. The integral against $\bigotimes_j\bigotimes_{D\in\mathcal A(W_j)}\widehat\nu^\circ_D$ of this character times $\prod_jw^\circ(W_j)$ is evaluated by Fubini one group at a time: integrating over the variables of $W_1$ with $h=P^\circ_{W_2}(a)\cdots P^\circ_{W_m}(a)$ fixed gives $a_{W_1}\chi(P^\circ_{W_2}(a)\cdots P^\circ_{W_m}(a))$ by \eqref{eq:part1-closed-schur}, and so on, ending with $\prod_ja_{W_j}\chi(\id)=\prod_ja_{W_j}$. Since $|\chi|\leq1$, $|a_W|\leq|w^\circ_\xi(W)|\prod_{D\in\mathcal A(W)}\|\widehat\nu^{\mathrm f,\circ}_{\xi,D}\|$, and the sum in \eqref{eq:part1-reference-gas} converges absolutely by the bound of Lemma~\ref{lem:grouped-source}.
\end{proof}

\begin{proposition}[joint extraction]\label{prop:joint-extraction}
For a compatible group family $\mathfrak B$ define the joint defect
\begin{equation}\label{eq:part1-joint-defect}
    R_\xi(\mathfrak B)=\sum_{\mathcal S\subseteq\mathfrak B}(-1)^{|\mathfrak B\setminus\mathcal S|}\Bigl(\prod_{W\in\mathfrak B\setminus\mathcal S}a_W\Bigr)A_\xi(\mathcal S)\;,\qquad R_\xi(\varnothing)=f_\xi(0)\;.
\end{equation}
Then, with $S^{\circ,[\mathfrak B]}_{\xi,C}$ the reference sum over group families compatible with every member of $\mathfrak B$,
\begin{equation}\label{eq:part1-scalar-extraction}
    F_\xi(0)=S^\circ_{\xi,C}f_\xi(0)+\sum_{\varnothing\ne\mathfrak B\text{ compatible}}S^{\circ,[\mathfrak B]}_{\xi,C}R_\xi(\mathfrak B)\;,
\end{equation}
and on every branch
\begin{equation}\label{eq:c-one-step-branch-identity}
    \mathcal J^{\mathrm a}_\xi[f_\xi]=e^{c_\xi}I_{\xi;1}\Bigl[S^\circ_{\xi,C}f_\xi(0)+\sum_{\varnothing\ne\mathfrak B\text{ compatible}}S^{\circ,[\mathfrak B]}_{\xi,C}R_\xi(\mathfrak B)+\mathcal B_\xi[f_\xi]\Bigr]\;.
\end{equation}
\end{proposition}

\begin{proof}
Boolean inversion of \eqref{eq:part1-joint-defect} gives $A_\xi(\mathfrak B)=\sum_{\mathcal S\subseteq\mathfrak B}(\prod_{\mathfrak B\setminus\mathcal S}a_W)R_\xi(\mathcal S)$; substitute into \eqref{eq:part1-grouped-source} and sum over $\mathfrak B\supseteq\mathcal S$: the families $\mathfrak B\setminus\mathcal S$ range over the group families compatible with every member of $\mathcal S$, giving $S^{\circ,[\mathcal S]}_{\xi,C}$. The interchange of the sums is justified by absolute convergence. Put $\beta_W=(|w_\xi(W)|+|w^\circ_\xi(W)|)\prod_{D\in\mathcal A(W)}\max(\|\widehat\nu^{\mathrm f}_{\xi,D}\|,\|\widehat\nu^{\mathrm f,\circ}_{\xi,D}\|)$; then $|a_W|\leq\beta_W$, $|A_\xi(\mathcal S)|\leq\|f_\xi\|_\infty\prod_{W\in\mathcal S}\beta_W$ and $|R_\xi(\mathcal S)|\leq\|f_\xi\|_\infty\prod_{W\in\mathcal S}2\beta_W$. The sum $\sum_W\beta_W$ over all groups is finite: a group is determined by its root family, its cluster list and its interval, there are finitely many root families at fixed volume, and $\sum_{W\, | \,\mathcal A(W)=\mathcal A}|w_\xi(W)|\leq\exp(\frac18\sum_{D\in\mathcal A}N_1(D))$ by Lemma~\ref{lem:vacuum-clusters}, likewise for $w^\circ_\xi$ by Lemma~\ref{lem:c2-reference}. A compatible group family has at most $N_{\mathrm m}$ members, the number of marked domains (its roots are pairwise distinct marked domains), so $\sum_{\mathfrak B}\sum_{\mathcal S\subseteq\mathfrak B}\prod_{W\in\mathfrak B\setminus\mathcal S}|a_W|\,|R_\xi(\mathcal S)|\leq\|f_\xi\|_\infty\sum_{\mathfrak B}\prod_{W\in\mathfrak B}3\beta_W\leq\|f_\xi\|_\infty\sum_{n\leq N_{\mathrm m}}\bigl(3\sum_W\beta_W\bigr)^n<\infty$; in particular $S^\circ_{\xi,C}$ and every $S^{\circ,[\mathcal S]}_{\xi,C}$ converge absolutely. Then \eqref{eq:c-physical-numerator} with $f=f_\xi$.
\end{proof}

At the reference data every nonempty defect vanishes: there $A^\circ_\xi(\mathcal S)=\prod_{W\in\mathcal S}a_W$ for every compatible $\mathcal S$ (proof of Lemma~\ref{lem:schur}), so $R^\circ_\xi(\mathfrak B)=\prod_{W\in\mathfrak B}a_W\sum_{\mathcal S\subseteq\mathfrak B}(-1)^{|\mathfrak B\setminus\mathcal S|}=0$ for $\mathfrak B\ne\varnothing$. The identity uses no division by a character value and no norm of an open transport.

\begin{lemma}[covered branches]\label{lem:covered}
Call $\xi$ covered when every bond of the contour is a bond of $\Omega_1$ and every free-input stencil $F_\gamma$ of the contour \eqref{eq:native-source-arc-stencils} lies in $I$, that is, $\mathcal F_r\subseteq I$ for every contour bond $r$, with $\mathcal F_r$ of \eqref{eq:regular-stencil-extension-support} (the convention $\mathcal F_r=\varnothing$ off $\Omega_1$ of Section~\ref{sec:joint-extraction} is not needed here, the first condition placing every contour bond in $\Omega_1$). On a covered branch $f_\xi(0)=W_\chi(C;U_1)$.
\end{lemma}

\begin{proof}
At $a=0$ the free vector is $B=\Psi_\xi(A_{\mathrm e})$, which vanishes on $I$. For a contour bond $r$, a bond of $\Omega_1$ on a covered branch, $U_\xi(0)(r)=e^{iq_U(B)(r)}U_1(r)$ by \eqref{eq:c-reconstructed-field}, and by Lemma~\ref{lem:row-locality} $q_U(B)(r)$ depends on $B|_{\mathcal F_r}$ only and vanishes when $B|_{\mathcal F_r}=0$ (for a free bond directly, for $r=b_0(c)$ because $\mathcal F_r=\mathcal E_c\cap E_*$). Since $\mathcal F_r\subseteq I$ on a covered branch, $U_\xi(0)=U_1$ on the contour.
\end{proof}
